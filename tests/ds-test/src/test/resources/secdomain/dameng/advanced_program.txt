[static_cursor_query]
sql:
declare
  cursor c_source is select id, name from app.program_source order by id;
  v_id int;
  v_name varchar(80);
begin
  open c_source;
  fetch c_source into v_id, v_name;
  close c_source;
end;
/
expect:
{
  "contains": [
    {"class":"RdbTableDomain", "sqlType":"SELECT", "auditKind":"QUERY", "schema":"app", "table":"program_source"}
  ]
}

[bulk_collect_query]
sql:
declare
  type row_list is table of app.program_source%rowtype;
  v_rows row_list;
begin
  select * bulk collect into v_rows from app.program_source order by id;
end;
/
expect:
{
  "contains": [
    {"class":"RdbTableDomain", "sqlType":"SELECT", "auditKind":"QUERY", "schema":"app", "table":"program_source"}
  ]
}

[forall_insert_and_cleanup]
sql:
declare
  type id_list is table of int index by int;
  v_ids id_list;
begin
  v_ids(1) := 201;
  forall i in 1 .. 1
    insert into app.program_target(id, name, amount) values(v_ids(i), 'FORALL', 1);
  delete from app.program_target where id = 201;
end;
/
expect:
{
  "contains": [
    {"class":"RdbInsertDomain", "sqlType":"INSERT", "auditKind":"DML", "schema":"app", "table":"program_target"},
    {"class":"RdbDeleteDomain", "sqlType":"DELETE", "auditKind":"DML", "schema":"app", "table":"program_target"}
  ]
}
----------

[dynamic_sql_and_static_delete]
sql:

  v_id int := 103;

  execute immediate 'delete from app.program_target where id = ?' using v_id;
  delete from app.program_target where id = v_id;
end;
/

  "contains": [
    {"class":"RdbDeleteDomain", "sqlType":"DELETE", "auditKind":"DML", "schema":"app", "table":"program_target"}
