[create_unqualified_procedure_with_dml]
sql:
create procedure refresh_employee(p_id int) authid current_user as
begin
  update employee set active = 1 where id = p_id;
end refresh_employee;
/
expect:
{
  "contains": [
    {"class":"RdbProcedureDomain", "sqlType":"CREATE_PROG_OBJ", "auditKind":"CREATE", "name":"refresh_employee"},
    {"class":"RdbUpdateDomain", "sqlType":"UPDATE", "auditKind":"DML", "table":"employee"}
  ]
}
----------
[create_function_with_query]
sql:
create function app.employee_name(p_id int) return varchar authid definer as
  v_name varchar(80);
begin
  select name into v_name from app.employee where id = p_id;
  return v_name;
end employee_name;
/
expect:
{
  "contains": [
    {"class":"RdbFunctionDomain", "sqlType":"CREATE_PROG_OBJ", "auditKind":"CREATE", "schema":"app", "name":"employee_name"},
    {"class":"RdbTableDomain", "sqlType":"SELECT", "auditKind":"QUERY", "schema":"app", "table":"employee"}
  ]
}
----------
