[create_schema_all_embedded_object_families]
sql:
create schema app authorization SYSDBA
create domain positive_id int check(value > 0)
create table base_table(id positive_id primary key, amount int)
create procedure add_row(p_id int) as begin insert into base_table values(p_id, 1); end;
create function plus_one(p_id int) return int as begin return p_id + 1; end;
create trigger trg_base after insert on base_table for each row begin update base_table set amount = amount + 1 where id = :new.id; end;
create package embed_pkg as function get_value return int; end embed_pkg;
create package body embed_pkg as function get_value return int as begin return 3; end; end embed_pkg;
create class embed_class as value int; member function get_value return int; end;
create class body embed_class as member function get_value return int as begin return value; end; end;
create link embed_link connect 'DPI' with bad_user identified by bad_password using '127.0.0.1:1';
create materialized view log on base_table with primary key
create materialized view mv_embed refresh complete as select id, amount from base_table
create operator @^ (function plus_one, leftarg int);
