[insert_on_duplicate_key_update]
sql:
INSERT INTO target_tab(id, value_col) VALUES (1, 'new') ON DUPLICATE KEY UPDATE value_col = VALUES(value_col);
expect:
{
  "contains": [
    {
      "class": "MyInsertDomain",
      "sqlTypes": ["MERGE"],
      "conflict": "UPDATE"
    }
  ]
}

{

    {
      "class": "MyReplaceDomain",
