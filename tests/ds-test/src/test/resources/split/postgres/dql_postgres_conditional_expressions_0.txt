SELECT a,
       CASE WHEN a = 1 THEN 'one'
            WHEN a = 2 THEN 'two'
            ELSE 'other'
       END AS searched_case
FROM (VALUES (1), (2), (3)) AS t(a);
SELECT a,
       CASE a WHEN 1 THEN 'one'
              WHEN 2 THEN 'two'
              ELSE 'other'
       END AS simple_case,
       CASE WHEN a <> 0 THEN 10 / a > 1 ELSE false END AS safe_predicate,
       CASE WHEN a > 10 THEN 'large' END AS omitted_else
FROM (VALUES (0), (2), (20)) AS t(a);
SELECT COALESCE(description, short_description, '(none)') AS display_text,
       NULLIF(value_text, '(none)') AS normalized_text
FROM (VALUES (NULL::text, 'short', '(none)'), ('full', 'short', 'full')) AS t(description, short_description, value_text);
SELECT NULLIF(1, 2.2) AS promoted_nullif,
       pg_typeof(NULLIF(1, 2.2)) AS promoted_type,
       GREATEST(1, NULL, 5, 3) AS greatest_ignores_null,
       LEAST(DATE '2024-01-03', NULL, DATE '2024-01-01') AS least_ignores_null;
SELECT label,
       CASE WHEN flag THEN GREATEST(score, fallback_score)
            ELSE LEAST(score, fallback_score)
       END AS conditional_extreme
FROM (VALUES ('a', true, 10, 7), ('b', false, 10, 7)) AS t(label, flag, score, fallback_score);
