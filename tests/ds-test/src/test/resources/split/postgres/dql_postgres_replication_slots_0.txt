SELECT pg_drop_replication_slot(slot_name)
FROM pg_replication_slots
WHERE slot_name IN ('split_phys_slot', 'split_phys_slot_copy', 'split_logical_slot', 'split_logical_slot_copy', 'split_bin_slot');
SELECT slot_name, lsn IS NOT NULL AS lsn_reserved FROM pg_create_physical_replication_slot('split_phys_slot', true, false);
SELECT slot_name, lsn IS NOT NULL AS copied_lsn_present FROM pg_copy_physical_replication_slot('split_phys_slot', 'split_phys_slot_copy', false);
SELECT slot_name, end_lsn IS NOT NULL AS physical_advanced FROM pg_replication_slot_advance('split_phys_slot_copy', pg_current_wal_lsn());
SELECT pg_drop_replication_slot('split_phys_slot_copy');
SELECT pg_drop_replication_slot('split_phys_slot');
SELECT slot_name, lsn IS NOT NULL AS logical_lsn_present FROM pg_create_logical_replication_slot('split_logical_slot', 'test_decoding', false);
SELECT pg_logical_emit_message(false, 'open_cdm_split', 'slot message text') IS NOT NULL AS emitted_text_message;
SELECT pg_logical_emit_message(false, 'open_cdm_split', E'\\x736c6f745f6d657373616765'::bytea) IS NOT NULL AS emitted_bytea_message;
SELECT count(*) >= 0 AS peek_text_changes FROM pg_logical_slot_peek_changes('split_logical_slot', NULL, NULL, 'include-xids', '0');
SELECT count(*) >= 0 AS get_text_changes FROM pg_logical_slot_get_changes('split_logical_slot', NULL, NULL, 'include-xids', '0');
SELECT count(*) = 0 AS no_text_changes_left FROM pg_logical_slot_get_changes('split_logical_slot', NULL, NULL, 'include-xids', '0');
SELECT slot_name, lsn IS NOT NULL AS logical_copy_lsn_present FROM pg_copy_logical_replication_slot('split_logical_slot', 'split_logical_slot_copy', false, 'test_decoding');
SELECT slot_name, end_lsn IS NOT NULL AS logical_advanced FROM pg_replication_slot_advance('split_logical_slot_copy', pg_current_wal_lsn());
SELECT pg_drop_replication_slot('split_logical_slot_copy');
SELECT pg_drop_replication_slot('split_logical_slot');
SELECT slot_name, lsn IS NOT NULL AS binary_slot_lsn_present FROM pg_create_logical_replication_slot('split_bin_slot', 'test_decoding', false);
SELECT pg_logical_emit_message(false, 'open_cdm_split', 'binary probe') IS NOT NULL AS emitted_binary_probe;
SELECT count(*) >= 0 AS peek_binary_changes FROM pg_logical_slot_peek_binary_changes('split_bin_slot', NULL, NULL);
SELECT count(*) >= 0 AS get_binary_changes FROM pg_logical_slot_get_binary_changes('split_bin_slot', NULL, NULL);
SELECT pg_drop_replication_slot('split_bin_slot');
