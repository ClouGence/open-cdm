SET search_path TO split_session_plan, public;
SET SESSION work_mem = '4MB';
SET LOCAL statement_timeout TO DEFAULT;
SET datestyle TO ISO, DMY;
SET TIME ZONE INTERVAL '-08:00' HOUR TO MINUTE;
SET TIME ZONE LOCAL;
SET NAMES 'UTF8';
SET XML OPTION DOCUMENT;
SHOW search_path;
SHOW TIME ZONE;
SHOW TRANSACTION ISOLATION LEVEL;
SHOW SESSION AUTHORIZATION;
SHOW ALL;
RESET search_path;
RESET TIME ZONE;
RESET TRANSACTION ISOLATION LEVEL;
RESET SESSION AUTHORIZATION;
RESET ALL;
PREPARE split_insert_plan (integer, text) AS INSERT INTO split_session_plan.plan_items VALUES ($1, $2) RETURNING id;
EXECUTE split_insert_plan(1, 'one');
PREPARE split_update_plan AS UPDATE split_session_plan.plan_items SET note = note || '_u' WHERE id = $1 RETURNING id, note;
EXECUTE split_update_plan(1);
PREPARE split_delete_plan (integer) AS DELETE FROM split_session_plan.plan_items WHERE id = $1 RETURNING id;
EXECUTE split_delete_plan(1);
PREPARE split_merge_plan (integer, text) AS MERGE INTO split_session_plan.plan_items t USING (VALUES ($1, $2)) AS s(id, note) ON t.id = s.id WHEN MATCHED THEN UPDATE SET note = s.note WHEN NOT MATCHED THEN INSERT (id, note) VALUES (s.id, s.note);
EXECUTE split_merge_plan(2, 'merged');
PREPARE split_values_plan AS VALUES (1, 'one'), (2, 'two');
EXECUTE split_values_plan;
CREATE TEMP TABLE split_values_copy AS EXECUTE split_values_plan WITH NO DATA;
DEALLOCATE split_insert_plan;
DEALLOCATE PREPARE split_update_plan;
DEALLOCATE ALL;
PREPARE split_cleanup_plan AS SELECT 1;
DEALLOCATE PREPARE ALL;
DISCARD TEMP;
DISCARD PLANS;
DISCARD SEQUENCES;
