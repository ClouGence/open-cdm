DROP SCHEMA IF EXISTS split_select_lock CASCADE;
CREATE SCHEMA split_select_lock;
CREATE TABLE split_select_lock.parent (
    id integer PRIMARY KEY,
    note text
);
CREATE TABLE split_select_lock.child (
    id integer PRIMARY KEY,
    parent_id integer REFERENCES split_select_lock.parent(id),
    note text
);
INSERT INTO split_select_lock.parent(id, note) VALUES (1, 'alpha'), (2, 'beta');
INSERT INTO split_select_lock.child(id, parent_id, note) VALUES (10, 1, 'child-alpha'), (20, 2, 'child-beta');
SELECT p.id FROM split_select_lock.parent AS p WHERE p.id = 1 FOR UPDATE;
SELECT p.id FROM split_select_lock.parent AS p ORDER BY p.id LIMIT 1 FOR NO KEY UPDATE OF p NOWAIT;
SELECT p.id, c.id FROM split_select_lock.parent AS p JOIN split_select_lock.child AS c ON c.parent_id = p.id FOR SHARE OF p SKIP LOCKED;
SELECT p.id, c.id FROM split_select_lock.parent AS p JOIN split_select_lock.child AS c ON c.parent_id = p.id FOR KEY SHARE OF p, c;
SELECT p.id FROM split_select_lock.parent AS p ORDER BY p.id FOR UPDATE OF p LIMIT 1;
SELECT p.id FROM split_select_lock.parent AS p ORDER BY p.id FETCH FIRST 1 ROW ONLY FOR SHARE SKIP LOCKED;
SELECT p.id, c.id FROM split_select_lock.parent AS p JOIN split_select_lock.child AS c ON c.parent_id = p.id FOR UPDATE OF p NOWAIT FOR SHARE OF c SKIP LOCKED;
SELECT * FROM (SELECT * FROM split_select_lock.parent FOR UPDATE) AS locked_parent WHERE id = 1;
TABLE split_select_lock.parent ORDER BY id LIMIT 1 FOR KEY SHARE;
DROP SCHEMA split_select_lock CASCADE;
