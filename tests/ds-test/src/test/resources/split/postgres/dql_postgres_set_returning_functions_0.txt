SELECT generate_series(1.1::numeric, 4::numeric, 1.3::numeric) AS n;
SELECT * FROM generate_series(TIMESTAMP '2008-03-01 00:00', TIMESTAMP '2008-03-04 12:00', INTERVAL '10 hours') AS g(ts);
SELECT * FROM generate_series(TIMESTAMPTZ '2001-10-22 00:00 -04:00', TIMESTAMPTZ '2001-11-01 00:00 -05:00', INTERVAL '1 day', 'America/New_York') AS g(ts);
SELECT a AS array_value, s AS subscript, a[s] AS value
FROM (VALUES (ARRAY[-1,-2]), (ARRAY[100,200,300])) AS arrays(a),
     LATERAL generate_subscripts(a, 1) AS g(s);
SELECT a AS array_value, s AS reverse_subscript, a[s] AS value
FROM (VALUES (ARRAY[10,20,30])) AS arrays(a),
     LATERAL generate_subscripts(a, 1, true) AS g(s);
SELECT * FROM unnest(ARRAY['a','b','c']) WITH ORDINALITY AS u(val, ord);
SELECT * FROM ROWS FROM(unnest(ARRAY[10,20], ARRAY['foo','bar','baz']), generate_series(101,102)) WITH ORDINALITY AS z(a,b,g,ord);
