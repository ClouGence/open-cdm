CREATE ROLE split_owned_source_a;
CREATE ROLE split_owned_source_b;
CREATE ROLE split_owned_receiver;
CREATE ROLE split_owned_session;
GRANT split_owned_receiver TO split_owned_session;
CREATE SCHEMA split_owned_schema AUTHORIZATION split_owned_source_a;
CREATE TABLE split_owned_schema.source_table (id integer);
ALTER TABLE split_owned_schema.source_table OWNER TO split_owned_source_b;
REASSIGN OWNED BY split_owned_source_a, split_owned_source_b TO split_owned_receiver;
DROP OWNED BY split_owned_source_a, split_owned_source_b RESTRICT;
SET SESSION AUTHORIZATION split_owned_session;
CREATE TABLE split_owned_schema.session_table (id integer);
REASSIGN OWNED BY CURRENT_USER TO split_owned_receiver;
DROP OWNED BY CURRENT_ROLE CASCADE;
DROP OWNED BY SESSION_USER RESTRICT;
RESET SESSION AUTHORIZATION;
DROP OWNED BY split_owned_receiver CASCADE;
DROP ROLE split_owned_source_a;
DROP ROLE split_owned_source_b;
DROP ROLE split_owned_session;
DROP ROLE split_owned_receiver;
