SELECT * FROM pg_available_wal_summaries();
SELECT tli, start_lsn, end_lsn FROM pg_available_wal_summaries() ORDER BY end_lsn DESC LIMIT 3;
SELECT summarized_tli, summarized_lsn, pending_lsn, summarizer_pid FROM pg_get_wal_summarizer_state();
WITH summary AS (
    SELECT tli, start_lsn, end_lsn
    FROM pg_available_wal_summaries()
    ORDER BY end_lsn DESC
    LIMIT 1
)
SELECT s.tli, s.start_lsn, s.end_lsn,
       c.relfilenode, c.reltablespace, c.reldatabase,
       c.relforknumber, c.relblocknumber, c.is_limit_block
FROM summary AS s
CROSS JOIN LATERAL pg_wal_summary_contents(s.tli, s.start_lsn, s.end_lsn) AS c
LIMIT 5;
SELECT COUNT(*) >= 0 AS wal_summary_contents_callable
FROM pg_available_wal_summaries() AS s
CROSS JOIN LATERAL pg_wal_summary_contents(s.tli, s.start_lsn, s.end_lsn) AS c;
