DROP SCHEMA IF EXISTS split_alter_table CASCADE;
CREATE SCHEMA split_alter_table;
CREATE TABLE split_alter_table.ref_table (id integer PRIMARY KEY);
INSERT INTO split_alter_table.ref_table VALUES (1), (2);
CREATE TABLE split_alter_table.base_table (
    id integer NOT NULL,
    code text NOT NULL,
    payload text,
    old_payload varchar(10),
    generated_col integer GENERATED ALWAYS AS (id * 2) STORED,
    identity_col integer GENERATED BY DEFAULT AS IDENTITY,
    parent_key integer,
    CONSTRAINT base_table_pk PRIMARY KEY (id)
);
INSERT INTO split_alter_table.base_table (id, code, payload, old_payload, parent_key) VALUES (1, 'a', 'payload', 'old', 1);
CREATE UNIQUE INDEX base_table_code_idx ON split_alter_table.base_table (code);
CREATE INDEX base_table_payload_idx ON split_alter_table.base_table (payload);
CREATE FUNCTION split_alter_table.touch_payload() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
    NEW.payload := COALESCE(NEW.payload, 'touched');
    RETURN NEW;
END
$$;
CREATE TRIGGER base_table_bi BEFORE INSERT ON split_alter_table.base_table FOR EACH ROW EXECUTE FUNCTION split_alter_table.touch_payload();
CREATE RULE base_table_notify AS ON INSERT TO split_alter_table.base_table DO ALSO NOTIFY split_alter_table_channel;
CREATE TABLE split_alter_table.inherit_parent (id integer, code text);
CREATE TABLE split_alter_table.inherit_child (id integer, code text, child_note text);
CREATE TYPE split_alter_table.typed_row AS (id integer, note text);
CREATE TABLE split_alter_table.typed_probe (id integer, note text);
CREATE TABLE split_alter_table.part_parent (region text NOT NULL, amount integer) PARTITION BY LIST (region);
