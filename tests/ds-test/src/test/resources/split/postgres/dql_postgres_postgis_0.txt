SELECT id, name FROM split_postgis.places
WHERE geom && ST_MakeEnvelope(-74.01, 40.70, -73.90, 40.80, 4326)
  AND ST_Intersects(area, ST_MakeEnvelope(-73.99, 40.74, -73.97, 40.76, 4326))
ORDER BY geom <-> ST_SetSRID(ST_MakePoint(-73.98, 40.75), 4326)
LIMIT 2;
SELECT id, ST_AsGeoJSON(ST_Transform(geom, 3857)), ST_Area(area::geography), ST_Distance(geog, ST_GeogFromText('SRID=4326;POINT(-73.985 40.748)'))
FROM split_postgis.places
ORDER BY id;
SELECT id, ST_Length(route::geography), ST_DWithin(geog, ST_GeogFromText('SRID=4326;POINT(-73.985 40.748)'), 1000)
FROM split_postgis.places
ORDER BY id;
