TRUNCATE split_cov.child_table, split_cov.parent_table RESTART IDENTITY CASCADE;
INSERT INTO split_cov.parent_table (id, code)
VALUES (1, 'a'), (2, 'b')
ON CONFLICT (id) DO UPDATE SET code = excluded.code
RETURNING id, code;
INSERT INTO split_cov.child_table (parent_id, mood, tags, doc, embedding)
VALUES (1, 'ok', ARRAY['a','b'], '{"kind":"demo","count":1}'::jsonb, '[1,2,3]')
RETURNING id, parent_id, embedding <-> '[1,1,1]'::vector AS distance;
UPDATE split_cov.child_table c
SET tags = tags || ARRAY['updated'],
    doc = jsonb_set(doc, '{count}', '2'::jsonb)
FROM split_cov.parent_table p
WHERE c.parent_id = p.id AND p.code = 'a'
RETURNING c.id, c.doc;
DELETE FROM split_cov.child_table c
USING split_cov.parent_table p
WHERE c.parent_id = p.id AND p.code = 'b'
RETURNING c.id;
MERGE INTO split_cov.parent_table p
USING (VALUES (1, 'aa'), (3, 'c')) AS s(id, code)
ON p.id = s.id
WHEN MATCHED THEN UPDATE SET code = s.code
WHEN NOT MATCHED THEN INSERT (id, code) VALUES (s.id, s.code);
COPY (SELECT id, code FROM split_cov.parent_table ORDER BY id) TO STDOUT WITH (FORMAT csv, HEADER true);
