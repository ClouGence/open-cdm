SELECT id, grp,
       sum(id) OVER w AS partition_sum,
       avg(id) OVER (w ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING) AS moving_avg
FROM (VALUES (1, 'a'), (2, 'a'), (3, 'a'), (4, 'b')) AS t(id, grp)
WINDOW w AS (PARTITION BY grp ORDER BY id)
ORDER BY grp, id;
SELECT id,
       count(*) OVER (ORDER BY id ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING EXCLUDE CURRENT ROW) AS others_count,
       sum(id) OVER (ORDER BY id ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING EXCLUDE NO OTHERS) AS nearby_sum
FROM (VALUES (1), (2), (3), (4)) AS t(id)
ORDER BY id;
SELECT id, bucket,
       count(*) OVER (ORDER BY bucket GROUPS BETWEEN 1 PRECEDING AND CURRENT ROW EXCLUDE GROUP) AS prior_groups,
       count(*) OVER (ORDER BY bucket GROUPS BETWEEN CURRENT ROW AND 1 FOLLOWING EXCLUDE TIES) AS current_next_ties
FROM (VALUES (1, 10), (2, 10), (3, 20), (4, 30)) AS t(id, bucket)
ORDER BY id;
SELECT ts, amount,
       sum(amount) OVER (ORDER BY ts RANGE BETWEEN INTERVAL '1 day' PRECEDING AND CURRENT ROW) AS one_day_sum,
       count(*) FILTER (WHERE amount > 10) OVER (ORDER BY ts ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW EXCLUDE CURRENT ROW) AS previous_large
FROM (VALUES (timestamp '2024-01-01 00:00', 5),
             (timestamp '2024-01-01 12:00', 15),
             (timestamp '2024-01-03 00:00', 20)) AS t(ts, amount)
ORDER BY ts;
SELECT id,
       row_number() OVER copied AS rn,
       last_value(id) OVER full_frame AS last_in_frame
FROM (VALUES (1), (2), (3)) AS t(id)
WINDOW base AS (ORDER BY id),
       copied AS (base ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW),
       full_frame AS (ORDER BY id ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING)
ORDER BY id;
