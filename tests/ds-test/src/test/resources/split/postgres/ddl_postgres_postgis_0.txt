DROP SCHEMA IF EXISTS split_postgis CASCADE;
CREATE SCHEMA split_postgis;
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE TABLE split_postgis.places (
    id bigserial PRIMARY KEY,
    name text,
    geom geometry(Point, 4326),
    area geometry(Polygon, 4326),
    route geometry(LineString, 4326),
    geog geography(Point, 4326)
);
CREATE INDEX idx_postgis_places_geom_gist ON split_postgis.places USING gist (geom);
CREATE INDEX idx_postgis_places_area_spgist ON split_postgis.places USING spgist (area);
CREATE INDEX idx_postgis_places_geog_gist ON split_postgis.places USING gist (geog);
CREATE INDEX idx_postgis_places_geom_3857_gist ON split_postgis.places USING gist (ST_Transform(geom, 3857));
DROP SCHEMA IF EXISTS split_postgis CASCADE;
