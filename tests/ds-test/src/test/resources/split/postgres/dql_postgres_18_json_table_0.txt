SELECT * FROM JSON_TABLE(
  jsonb '{"favorites":[{"kind":"comedy","films":[{"title":"Bananas","director":"Woody Allen"},{"title":"The Dinner Game","director":"Francis Veber"}]},{"kind":"horror","films":[{"title":"Psycho","director":"Alfred Hitchcock"}]}]}',
  '$.favorites[*]'
  COLUMNS (
    id FOR ORDINALITY,
    kind text PATH '$.kind',
    title jsonb PATH '$.films[*].title' WITH WRAPPER,
    director jsonb PATH '$.films[*].director' WITH WRAPPER
  )
) AS jt;
SELECT * FROM JSON_TABLE(
  jsonb '{"favorites":[{"kind":"horror","films":[{"title":"Psycho","director":"Alfred Hitchcock"}]},{"kind":"thriller","films":[{"title":"Vertigo","director":"Alfred Hitchcock"}]}]}',
  '$.favorites[*] ? (@.films[*].director == $filter)'
  PASSING 'Alfred Hitchcock' AS filter
  COLUMNS (
    id FOR ORDINALITY,
    kind text PATH '$.kind',
    title text FORMAT JSON PATH '$.films[*].title' OMIT QUOTES,
    director text PATH '$.films[*].director' KEEP QUOTES
  )
) AS jt;
SELECT * FROM JSON_TABLE(
  jsonb '{"favorites":[{"kind":"comedy","films":[{"title":"Bananas","director":"Woody Allen"},{"title":"The Dinner Game","director":"Francis Veber"}]},{"kind":"horror","films":[{"title":"Psycho","director":"Alfred Hitchcock"}]}]}',
  '$.favorites[*]'
  COLUMNS (
    id FOR ORDINALITY,
    kind text PATH '$.kind',
    NESTED PATH '$.films[*]' COLUMNS (
      film_id FOR ORDINALITY,
      title text FORMAT JSON PATH '$.title' OMIT QUOTES,
      director text PATH '$.director' KEEP QUOTES
    )
  )
) AS jt;
SELECT * FROM JSON_TABLE(
  jsonb '{"a":1}',
  '$'
  COLUMNS (
    id FOR ORDINALITY,
    has_a boolean EXISTS PATH '$.a',
    has_b int EXISTS PATH 'strict $.b' UNKNOWN ON ERROR,
    missing int PATH '$.b' DEFAULT 42 ON EMPTY DEFAULT 7 ON ERROR
  )
) AS jt;
SELECT * FROM JSON_TABLE(
  jsonb '{"a":[1,2,3]}',
  'strict $.missing[*]'
  COLUMNS (item int PATH '$')
  EMPTY ON ERROR
) AS jt;
SELECT * FROM JSON_TABLE(
  '{"favorites":{"movies":[{"name":"One","director":"John Doe"}],"books":[{"name":"Mystery","authors":[{"name":"Brown Dan"},{"name":"Ada Blue"}]}]}}'::json,
  '$.favorites'
  COLUMNS (
    user_id FOR ORDINALITY,
    NESTED '$.movies[*]' COLUMNS (
      movie_id FOR ORDINALITY,
      mname text PATH '$.name',
      director text
    ),
    NESTED '$.books[*]' COLUMNS (
      book_id FOR ORDINALITY,
      bname text PATH '$.name',
      NESTED '$.authors[*]' COLUMNS (
        author_id FOR ORDINALITY,
        author_name text PATH '$.name'
      )
    )
  )
) AS jt;
