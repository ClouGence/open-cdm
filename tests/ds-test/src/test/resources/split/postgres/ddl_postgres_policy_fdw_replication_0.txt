CREATE EXTENSION IF NOT EXISTS file_fdw;
CREATE EXTENSION IF NOT EXISTS postgres_fdw;
ALTER TABLE split_prog.accounts ENABLE ROW LEVEL SECURITY;
CREATE POLICY accounts_owner_policy ON split_prog.accounts
AS PERMISSIVE FOR SELECT TO PUBLIC
USING (owner_name = current_user);
ALTER POLICY accounts_owner_policy ON split_prog.accounts RENAME TO accounts_select_policy;
ALTER POLICY accounts_select_policy ON split_prog.accounts USING (id > 0);
CREATE TEXT SEARCH CONFIGURATION split_prog.english_copy (COPY = pg_catalog.english);
ALTER TEXT SEARCH CONFIGURATION split_prog.english_copy ALTER MAPPING FOR asciiword WITH english_stem;
CREATE SERVER split_file_server FOREIGN DATA WRAPPER file_fdw;
CREATE FOREIGN TABLE split_prog.foreign_csv (id int, owner_name text)
SERVER split_file_server OPTIONS (filename '/tmp/split_prog.csv', format 'csv', header 'true');
CREATE SERVER split_pg_server FOREIGN DATA WRAPPER postgres_fdw OPTIONS (host '127.0.0.1', dbname 'postgres', port '5432');
CREATE USER MAPPING FOR CURRENT_USER SERVER split_pg_server OPTIONS (user 'postgres');
CREATE PUBLICATION split_pub FOR TABLE split_prog.accounts WITH (publish = 'insert, update, delete, truncate');
CREATE SUBSCRIPTION split_sub CONNECTION 'host=127.0.0.1 port=5432 dbname=postgres user=postgres' PUBLICATION split_pub WITH (connect = false, enabled = false);
