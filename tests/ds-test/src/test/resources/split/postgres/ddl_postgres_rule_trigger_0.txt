DROP EVENT TRIGGER IF EXISTS split_event_probe;
DROP SCHEMA IF EXISTS split_trigger_probe CASCADE;
CREATE SCHEMA split_trigger_probe;
CREATE TABLE split_trigger_probe.accounts (
    id integer PRIMARY KEY,
    balance integer,
    note text
);
CREATE TABLE split_trigger_probe.account_audit (
    id integer,
    action text,
    old_balance integer,
    new_balance integer
);
CREATE VIEW split_trigger_probe.account_view AS
    SELECT id, balance, note FROM split_trigger_probe.accounts;
CREATE FUNCTION split_trigger_probe.audit_row()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO split_trigger_probe.account_audit(id, action, new_balance)
        VALUES (NEW.id, TG_ARGV[0], NEW.balance);
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO split_trigger_probe.account_audit(id, action, old_balance, new_balance)
        VALUES (NEW.id, TG_ARGV[0], OLD.balance, NEW.balance);
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO split_trigger_probe.account_audit(id, action, old_balance)
        VALUES (OLD.id, TG_ARGV[0], OLD.balance);
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$;
CREATE FUNCTION split_trigger_probe.audit_statement()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    INSERT INTO split_trigger_probe.account_audit(id, action)
    VALUES (0, TG_ARGV[0]);
    RETURN NULL;
END;
$$;
CREATE FUNCTION split_trigger_probe.view_insert_row()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    INSERT INTO split_trigger_probe.accounts(id, balance, note)
    VALUES (NEW.id, NEW.balance, NEW.note);
    RETURN NEW;
END;
$$;
CREATE FUNCTION split_trigger_probe.event_logger()
RETURNS event_trigger
LANGUAGE plpgsql
AS $$
BEGIN
    NULL;
END;
$$;
CREATE OR REPLACE TRIGGER accounts_check_update
    BEFORE UPDATE OF balance ON split_trigger_probe.accounts
    FOR EACH ROW
    WHEN (OLD.balance IS DISTINCT FROM NEW.balance)
    EXECUTE FUNCTION split_trigger_probe.audit_row('before_update_balance');
CREATE TRIGGER accounts_insert_stmt
    AFTER INSERT ON split_trigger_probe.accounts
    REFERENCING NEW TABLE AS inserted_rows
    FOR EACH STATEMENT
    EXECUTE PROCEDURE split_trigger_probe.audit_statement('after_insert_statement');
CREATE TRIGGER accounts_update_transition
    AFTER UPDATE ON split_trigger_probe.accounts
    REFERENCING OLD TABLE AS old_rows NEW TABLE AS new_rows
    FOR EACH STATEMENT
    EXECUTE FUNCTION split_trigger_probe.audit_statement('after_update_statement');
CREATE CONSTRAINT TRIGGER accounts_balance_constraint
    AFTER INSERT OR UPDATE ON split_trigger_probe.accounts
    DEFERRABLE INITIALLY DEFERRED
    FOR EACH ROW
    WHEN (NEW.balance >= 0)
    EXECUTE FUNCTION split_trigger_probe.audit_row('constraint_balance');
CREATE TRIGGER account_view_insert
    INSTEAD OF INSERT ON split_trigger_probe.account_view
    FOR EACH ROW
    EXECUTE FUNCTION split_trigger_probe.view_insert_row();
COMMENT ON TRIGGER accounts_check_update ON split_trigger_probe.accounts IS 'balance update trigger';
ALTER TRIGGER accounts_check_update ON split_trigger_probe.accounts RENAME TO accounts_check_update_renamed;
ALTER TRIGGER accounts_check_update_renamed ON split_trigger_probe.accounts DEPENDS ON EXTENSION plpgsql;
ALTER TRIGGER accounts_check_update_renamed ON split_trigger_probe.accounts NO DEPENDS ON EXTENSION plpgsql;
CREATE RULE account_insert_skip AS ON INSERT TO split_trigger_probe.account_audit
    WHERE NEW.action = 'skip'
    DO INSTEAD NOTHING;
CREATE OR REPLACE RULE account_update_notify AS ON UPDATE TO split_trigger_probe.account_audit
    DO ALSO NOTIFY split_trigger_probe_channel;
CREATE RULE account_delete_multi AS ON DELETE TO split_trigger_probe.account_audit
    DO ALSO (
        NOTIFY split_trigger_probe_channel;
        INSERT INTO split_trigger_probe.account_audit(id, action) VALUES (OLD.id, 'deleted')
    );
COMMENT ON RULE account_update_notify ON split_trigger_probe.account_audit IS 'rule notify probe';
ALTER RULE account_update_notify ON split_trigger_probe.account_audit RENAME TO account_update_notify_renamed;
CREATE EVENT TRIGGER split_event_probe
    ON ddl_command_end
    WHEN TAG IN ('CREATE TABLE', 'ALTER TABLE')
    EXECUTE FUNCTION split_trigger_probe.event_logger();
CREATE EVENT TRIGGER split_event_probe_proc
    ON sql_drop
    EXECUTE PROCEDURE split_trigger_probe.event_logger();
COMMENT ON EVENT TRIGGER split_event_probe IS 'event trigger probe';
ALTER EVENT TRIGGER split_event_probe DISABLE;
ALTER EVENT TRIGGER split_event_probe ENABLE REPLICA;
ALTER EVENT TRIGGER split_event_probe ENABLE ALWAYS;
ALTER EVENT TRIGGER split_event_probe OWNER TO CURRENT_USER;
ALTER EVENT TRIGGER split_event_probe RENAME TO split_event_probe_renamed;
DROP EVENT TRIGGER IF EXISTS split_event_probe_renamed;
DROP EVENT TRIGGER IF EXISTS split_event_probe_proc;
DROP RULE IF EXISTS account_insert_skip ON split_trigger_probe.account_audit;
DROP RULE IF EXISTS account_update_notify_renamed ON split_trigger_probe.account_audit CASCADE;
DROP RULE IF EXISTS account_delete_multi ON split_trigger_probe.account_audit RESTRICT;
DROP TRIGGER IF EXISTS accounts_check_update_renamed ON split_trigger_probe.accounts;
DROP TRIGGER IF EXISTS accounts_insert_stmt ON split_trigger_probe.accounts CASCADE;
DROP TRIGGER IF EXISTS accounts_update_transition ON split_trigger_probe.accounts RESTRICT;
DROP TRIGGER IF EXISTS accounts_balance_constraint ON split_trigger_probe.accounts;
DROP TRIGGER IF EXISTS account_view_insert ON split_trigger_probe.account_view;
DROP SCHEMA split_trigger_probe CASCADE;
