WITH RECURSIVE graph(f, t, label) AS (
    VALUES (1, 2, 'arc 1 -> 2'),
           (1, 3, 'arc 1 -> 3'),
           (2, 4, 'arc 2 -> 4'),
           (4, 1, 'arc 4 -> 1')
), search_graph(f, t, label, depth) AS (
    SELECT f, t, label, 1 FROM graph WHERE f = 1
    UNION ALL
    SELECT g.f, g.t, g.label, sg.depth + 1 FROM graph g JOIN search_graph sg ON g.f = sg.t WHERE sg.depth < 3
) SEARCH DEPTH FIRST BY f, t SET seq
SELECT f, t, label FROM search_graph ORDER BY seq;

WITH RECURSIVE graph(f, t, label) AS (
    VALUES (1, 2, 'arc 1 -> 2'),
           (1, 3, 'arc 1 -> 3'),
           (2, 4, 'arc 2 -> 4'),
           (4, 1, 'arc 4 -> 1')
), search_graph(f, t, label) AS (
    SELECT * FROM graph WHERE f = 1
    UNION ALL
    SELECT g.* FROM graph g JOIN search_graph sg ON g.f = sg.t
) CYCLE f, t SET is_cycle USING path
SELECT f, t, label, is_cycle FROM search_graph ORDER BY path LIMIT 8;
