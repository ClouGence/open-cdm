SELECT DATE '2001-09-28' + 7 AS date_plus_int,
       7 + DATE '2001-09-28' AS int_plus_date,
       DATE '2001-09-28' + INTERVAL '1 hour' AS date_plus_interval,
       DATE '2001-09-28' + TIME '03:00' AS date_plus_time,
       DATE '2001-10-01' - DATE '2001-09-28' AS date_minus_date,
       DATE '2001-10-01' - 7 AS date_minus_int,
       DATE '2001-09-28' - INTERVAL '1 hour' AS date_minus_interval;
SELECT INTERVAL '1 day' + INTERVAL '1 hour' AS interval_plus_interval,
       - INTERVAL '23 hours' AS negated_interval,
       INTERVAL '1 day' - INTERVAL '1 hour' AS interval_minus_interval,
       INTERVAL '1 second' * 900 AS interval_times_scalar,
       3.5 * INTERVAL '1 hour' AS scalar_times_interval,
       INTERVAL '1 hour' / 1.5 AS interval_div_scalar;
SELECT TIMESTAMP '2001-09-28 01:00' + INTERVAL '23 hours' AS timestamp_plus_interval,
       TIMESTAMP '2001-09-28 23:00' - INTERVAL '23 hours' AS timestamp_minus_interval,
       TIMESTAMP '2001-09-29 03:00' - TIMESTAMP '2001-07-27 12:00' AS timestamp_minus_timestamp,
       TIME '01:00' + INTERVAL '3 hours' AS time_plus_interval,
       TIME '05:00' - TIME '03:00' AS time_minus_time,
       TIME '05:00' - INTERVAL '2 hours' AS time_minus_interval;
SELECT age(TIMESTAMP '2001-04-10', TIMESTAMP '1957-06-13') AS age_between,
       age(TIMESTAMP '1957-06-13') AS age_from_current_date,
       date_add('2021-10-31 00:00:00+02'::timestamptz, '1 day'::interval, 'Europe/Warsaw') AS add_in_zone,
       date_add('2021-10-31 00:00:00+02'::timestamptz, '1 day'::interval) AS add_in_session_zone,
       date_subtract('2021-11-01 00:00:00+01'::timestamptz, '1 day'::interval, 'Europe/Warsaw') AS subtract_in_zone,
       date_subtract('2021-11-01 00:00:00+01'::timestamptz, '1 day'::interval) AS subtract_in_session_zone;
SELECT date_bin('15 minutes', TIMESTAMP '2001-02-16 20:38:40', TIMESTAMP '2001-02-16 20:05:00') AS bin_timestamp,
       date_bin('5 min'::interval, TIMESTAMP '2020-02-01 01:01:01', TIMESTAMP '2020-02-01 00:02:30') AS shifted_bin_timestamp,
       date_bin('30 minutes'::interval, TIMESTAMPTZ '2024-02-01 15:00:00+00', TIMESTAMPTZ '2024-02-01 17:00:00+00') AS bin_timestamptz;
SELECT date_part('hour', TIMESTAMP '2001-02-16 20:38:40') AS part_hour,
       date_part('month', INTERVAL '2 years 3 months') AS part_interval_month,
       date_part('timezone_hour', TIME WITH TIME ZONE '13:30:25.575401-04') AS part_timezone_hour;
SELECT date_trunc('hour', TIMESTAMP '2001-02-16 20:38:40') AS trunc_hour,
       date_trunc('year', TIMESTAMP '2001-02-16 20:38:40') AS trunc_year,
       date_trunc('day', TIMESTAMP WITH TIME ZONE '2001-02-16 20:38:40+00') AS trunc_timestamptz_day,
       date_trunc('day', TIMESTAMP WITH TIME ZONE '2001-02-16 20:38:40+00', 'Australia/Sydney') AS trunc_timestamptz_zone,
       date_trunc('hour', INTERVAL '3 days 02:47:33') AS trunc_interval;
SELECT EXTRACT(CENTURY FROM TIMESTAMP '2000-12-16 12:21:13') AS century_ad,
       EXTRACT(CENTURY FROM DATE '0001-12-31 BC') AS century_bc,
       EXTRACT(DAY FROM INTERVAL '40 days 1 minute') AS interval_day,
       EXTRACT(DECADE FROM TIMESTAMP '2001-02-16 20:38:40') AS decade,
       EXTRACT(DOW FROM TIMESTAMP '2001-02-16 20:38:40') AS dow,
       EXTRACT(DOY FROM TIMESTAMP '2001-02-16 20:38:40') AS doy;
SELECT EXTRACT(EPOCH FROM TIMESTAMP WITH TIME ZONE '2001-02-16 20:38:40.12-08') AS epoch_timestamptz,
       EXTRACT(EPOCH FROM INTERVAL '5 days 3 hours') AS epoch_interval,
       EXTRACT(HOUR FROM TIME '17:12:28.5') AS hour,
       EXTRACT(ISODOW FROM TIMESTAMP '2001-02-18 20:38:40') AS isodow,
       EXTRACT(ISOYEAR FROM DATE '2006-01-01') AS isoyear,
       EXTRACT(JULIAN FROM TIMESTAMP '2006-01-01 12:00') AS julian;
SELECT EXTRACT(MICROSECONDS FROM TIME '17:12:28.5') AS microseconds,
       EXTRACT(MILLISECONDS FROM TIME '17:12:28.5') AS milliseconds,
       EXTRACT(MILLENNIUM FROM INTERVAL '2001 years') AS millennium,
       EXTRACT(MINUTE FROM TIMESTAMP '2001-02-16 20:38:40') AS minute,
       EXTRACT(MONTH FROM INTERVAL '2 years 13 months') AS month,
       EXTRACT(QUARTER FROM TIMESTAMP '2001-02-16 20:38:40') AS quarter,
       EXTRACT(SECOND FROM TIME '17:12:28.5') AS second;
SELECT EXTRACT(TIMEZONE FROM TIMESTAMP WITH TIME ZONE '2001-02-16 20:38:40.12-08') AS timezone_seconds,
       EXTRACT(TIMEZONE_HOUR FROM TIME WITH TIME ZONE '17:12:28.5-08') AS timezone_hour,
       EXTRACT(TIMEZONE_MINUTE FROM TIME WITH TIME ZONE '17:12:28.5+05:30') AS timezone_minute,
       EXTRACT(WEEK FROM TIMESTAMP '2001-02-16 20:38:40') AS week,
       EXTRACT(YEAR FROM TIMESTAMP '2001-02-16 20:38:40') AS year;
SELECT isfinite(DATE '2001-02-16') AS finite_date,
       isfinite(TIMESTAMP 'infinity') AS finite_timestamp,
       isfinite(INTERVAL '4 hours') AS finite_interval,
       justify_days(INTERVAL '1 year 65 days') AS justified_days,
       justify_hours(INTERVAL '50 hours 10 minutes') AS justified_hours,
       justify_interval(INTERVAL '1 mon -1 hour') AS justified_interval;
SELECT make_date(2013, 7, 15) AS made_date,
       make_interval(years => 1, months => 2, weeks => 3, days => 4, hours => 5, mins => 6, secs => 7.5) AS made_interval,
       make_interval(days => 10) AS made_interval_days,
       make_time(8, 15, 23.5) AS made_time,
       make_timestamp(2013, 7, 15, 8, 15, 23.5) AS made_timestamp,
       make_timestamptz(2013, 7, 15, 8, 15, 23.5, 'America/New_York') AS made_timestamptz;
SELECT now() AS now_value,
       statement_timestamp() AS statement_ts,
       transaction_timestamp() AS transaction_ts,
       clock_timestamp() AS clock_ts,
       timeofday() AS timeofday_text,
       to_timestamp(1284352323) AS epoch_to_timestamptz;
SELECT CURRENT_DATE AS current_date_value,
       CURRENT_TIME AS current_time_value,
       CURRENT_TIME(2) AS current_time_2,
       CURRENT_TIMESTAMP AS current_timestamp_value,
       CURRENT_TIMESTAMP(0) AS current_timestamp_0,
       LOCALTIME AS localtime_value,
       LOCALTIME(0) AS localtime_0,
       LOCALTIMESTAMP AS localtimestamp_value,
       LOCALTIMESTAMP(2) AS localtimestamp_2,
       TIMESTAMP 'now' AS timestamp_now_literal;
SELECT TIMESTAMP '2001-02-16 20:38:40' AT TIME ZONE 'America/Denver' AS timestamp_at_zone,
       TIMESTAMP WITH TIME ZONE '2001-02-16 20:38:40-05' AT TIME ZONE 'America/Denver' AS timestamptz_at_zone,
       TIMESTAMP '2001-02-16 20:38:40' AT TIME ZONE 'Asia/Tokyo' AT TIME ZONE 'America/Chicago' AS chained_at_time_zone,
       TIMESTAMP WITH TIME ZONE '2001-02-16 20:38:40-05' AT TIME ZONE '+05' AS posix_zone_name,
       TIME WITH TIME ZONE '05:34:17-05' AT TIME ZONE 'UTC' AS timetz_at_zone,
       TIME WITH TIME ZONE '05:34:17-05' AT TIME ZONE INTERVAL '-08:00' AS timetz_at_interval;
SELECT timezone('America/Denver', TIMESTAMP '2001-02-16 20:38:40') AS timezone_timestamp,
       timezone('America/Denver', TIMESTAMP WITH TIME ZONE '2001-02-16 20:38:40-05') AS timezone_timestamptz,
       timezone('UTC', TIME WITH TIME ZONE '05:34:17-05') AS timezone_timetz;
SELECT (DATE '2001-10-29', DATE '2001-10-30') OVERLAPS (DATE '2001-10-30', DATE '2001-10-31') AS touching_periods_do_not_overlap,
       (DATE '2001-10-30', DATE '2001-10-30') OVERLAPS (DATE '2001-10-30', DATE '2001-10-31') AS equal_endpoint_overlaps,
       (TIMESTAMP '2001-02-16 20:00', INTERVAL '2 hours') OVERLAPS (TIMESTAMP '2001-02-16 21:00', INTERVAL '2 hours') AS timestamp_interval_overlaps;
SELECT pg_sleep(0);
SELECT pg_sleep_for(INTERVAL '0 seconds');
SELECT pg_sleep_until(clock_timestamp());
