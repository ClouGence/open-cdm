SELECT xmlelement(name foo, xmlattributes('xyz' AS bar), xmlcomment('note'), xmlforest(123 AS n)) AS made_xml;
SELECT xmlconcat(XMLPARSE(CONTENT '<a/>'), XMLPARSE(CONTENT '<b>c</b>')) AS concatenated_xml;
SELECT XMLSERIALIZE(CONTENT XMLPARSE(CONTENT 'abc<foo>bar</foo>') AS text NO INDENT) AS serialized_xml;
SELECT XMLROOT(XMLPARSE(DOCUMENT '<content>abc</content>'), VERSION '1.0', STANDALONE YES) AS rooted_xml;
SELECT XMLEXISTS('//town[text() = ''Toronto'']' PASSING BY VALUE XMLPARSE(DOCUMENT '<towns><town>Toronto</town><town>Ottawa</town></towns>')) AS has_toronto;
SELECT *
FROM XMLTABLE('/ROWS/ROW'
       PASSING XMLPARSE(DOCUMENT '<ROWS><ROW><COUNTRY_NAME>Japan</COUNTRY_NAME><REGION_ID>3</REGION_ID></ROW><ROW><COUNTRY_NAME>India</COUNTRY_NAME></ROW></ROWS>')
       COLUMNS id FOR ORDINALITY,
               country text PATH 'COUNTRY_NAME' NOT NULL,
               region_id int PATH 'REGION_ID' DEFAULT 0);
SELECT *
FROM XMLTABLE(XMLNAMESPACES('http://x.y' AS zz),
       '/zz:rows/zz:row'
       PASSING XMLPARSE(DOCUMENT '<zz:rows xmlns:zz="http://x.y"><zz:row><zz:name>alice</zz:name></zz:row></zz:rows>')
       COLUMNS ord FOR ORDINALITY,
               name text PATH 'zz:name' DEFAULT 'missing');
