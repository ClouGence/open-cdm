WITH RECURSIVE nums(n) AS (
    VALUES (1)
    UNION ALL
    SELECT n + 1 FROM nums WHERE n < 3
), ranked AS (
    SELECT n, sum(n) OVER (ORDER BY n ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_sum
    FROM nums
)
SELECT n, running_sum, count(*) FILTER (WHERE n % 2 = 1) OVER () AS odd_count
FROM ranked
ORDER BY n;
SELECT p.id, p.code, child.child_id
FROM split_cov.parent_table p
LEFT JOIN LATERAL (
    SELECT c.id AS child_id
    FROM split_cov.child_table c
    WHERE c.parent_id = p.id
    ORDER BY c.id
    FETCH FIRST 1 ROW WITH TIES
) child ON true
WHERE EXISTS (SELECT 1 FROM split_cov.parent_table p2 WHERE p2.id = p.id);
SELECT t.id, t.val, t.g, t.ord
FROM ROWS FROM (
    jsonb_to_recordset('[{"id":1,"val":"a"},{"id":2,"val":"b"}]'::jsonb) AS (id int, val text),
    generate_series(10, 11)
) WITH ORDINALITY AS t(id, val, g, ord);
SELECT GROUPING(code) AS grp, code, count(*) FILTER (WHERE id > 0) AS total
FROM split_cov.parent_table
GROUP BY GROUPING SETS ((code), ());
SELECT id, code
FROM split_cov.parent_table TABLESAMPLE SYSTEM (100) REPEATABLE (42)
ORDER BY id;
SELECT ROW(1, 'a') IS DISTINCT FROM ROW(2, 'b') AS row_diff,
       1 = ANY (ARRAY[1,2]) AS any_match,
       3 > ALL (ARRAY[1,2]) AS all_match,
       'abc' COLLATE "C" = 'abc' AS collated_match;
SELECT doc @? '$.kind ? (@ == "demo")' AS has_demo_kind
FROM split_cov.child_table;
