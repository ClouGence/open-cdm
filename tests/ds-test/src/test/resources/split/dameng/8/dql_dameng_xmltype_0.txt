SELECT ID, DOC FROM XMLTYPE_DOCS ORDER BY ID;
SELECT ID, XMLQUERY(DOC, '/ROOT/ITEM') AS ITEM_XML FROM XMLTYPE_DOCS WHERE ID IN(1,2) ORDER BY ID;
SELECT ID, EXISTSNODE(DOC, '/ROOT/ITEM') AS HAS_ITEM FROM XMLTYPE_DOCS WHERE ID IN(1,2,4) ORDER BY ID;
SELECT ID, EXTRACTVALUE(DOC, '/ROOT/ITEM') AS ITEM_VALUE FROM XMLTYPE_DOCS WHERE ID IN(1,2) ORDER BY ID;
SELECT APPENDCHILDXML(DOC, '/ROOT', '<TAIL>Z</TAIL>') FROM XMLTYPE_DOCS WHERE ID=1;
SELECT DELETEXML(DOC, '/ROOT/ITEM') FROM XMLTYPE_DOCS WHERE ID=1;
SELECT UPDATEXML(DOC, '/ROOT/ITEM', '<ITEM ID="9">Z</ITEM>') FROM XMLTYPE_DOCS WHERE ID=1;
SELECT D.ID, X.ITEM_ID, X.ITEM_TEXT FROM XMLTYPE_DOCS D, XMLTABLE('/ROOT/ITEM' PASSING D.DOC COLUMNS ITEM_ID INT PATH '@ID', ITEM_TEXT VARCHAR2(20) PATH 'text()') X WHERE D.ID IN(1,2) ORDER BY D.ID;
SELECT X.COLUMN_VALUE FROM XMLTYPE_DOCS D, TABLE(XMLSEQUENCE(EXTRACT(D.DOC, '/ROOT/ITEM'))) X WHERE D.ID=1;
SELECT XMLAGG(DOC ORDER BY ID) FROM XMLTYPE_DOCS WHERE ID IN(5,6);
SELECT COLUMN_NAME, DATA_TYPE FROM USER_TAB_COLUMNS WHERE TABLE_NAME='XMLTYPE_DOCS' ORDER BY COLUMN_ID;
SELECT ID, T.VALUE.NAME AS NAME_VALUE, T.VALUE.DOC AS DOC_VALUE FROM XML_HOLDER_TAB T;
SELECT ID, DOC FROM XMLTYPE_DOCS WHERE ID=10;
SELECT ID FROM XMLTYPE_DOCS WHERE DOC=XMLTYPE('<ROOT/>');
SELECT XMLQUERY(DOC, NULL) FROM XMLTYPE_DOCS WHERE ID=1;
SELECT EXTRACTVALUE(DOC, '/ROOT/*') FROM XMLTYPE_DOCS WHERE ID=1;
