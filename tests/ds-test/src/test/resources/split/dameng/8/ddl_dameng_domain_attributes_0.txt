  expr_value dom_attr_expr,
  multi_value dom_attr_multi,
  named_value dom_attr_named,
  max_name_value DOM_AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA
);
insert into domain_attr_use(id, multi_value) values(1, 5);

from domain_attr_use where id = 1;
update domain_attr_use set multi_value = 9, named_value = 1 where id = 1;

  id int primary key,
  inherited_value dom_attr_expr,
  overridden_value dom_attr_expr default 8

and name in ('DOM_ATTR_NULL', 'DOM_ATTR_EXPR', 'DOM_ATTR_MULTI', 'DOM_ATTR_NAMED');
