create table app.bin_t(
  id int primary key,c_binary binary(4) default 0xAA not null,
  c_varbinary varbinary(8) default 0xBB,c_raw raw(8) default 0xCC,
  c_null raw(8) null,constraint ck_bin check(utl_raw.length(c_varbinary)<=8),
  constraint uk_bin unique(c_varbinary,c_raw));
comment on column app.bin_t.c_binary is 'fixed binary';
comment on column app.bin_t.c_raw is 'raw variable binary';
create index app.idx_bin on app.bin_t(c_varbinary,c_binary);
alter table app.bin_t add c_alter raw(4) default 0xDD;
alter table app.bin_t modify c_alter raw(8);
insert into app.bin_t values
  (1,0x2A3B,0x414243,hextoraw('444546'),null,0x47);
insert into app.bin_t(id,c_binary,c_varbinary,c_raw,c_null)
  values(2,hextoraw('0102'),utl_raw.cast_to_raw('var'),
  utl_raw.cast_to_raw('raw'),hextoraw('00FF'));
insert into app.bin_t(id) values(3);
update app.bin_t set
  c_varbinary=utl_raw.concat(c_varbinary,hextoraw('21')),
  c_raw=utl_raw.reverse_raw(c_raw),c_alter=hextoraw('ABCD') where id=2;
select id,utl_raw.length(c_binary),utl_raw.length(c_varbinary),
  utl_raw.length(c_raw),rawtohex(c_binary),rawtohex(c_varbinary),rawtohex(c_raw)
  from app.bin_t order by id;
select rawtohex(utl_raw.substr(c_varbinary,2,2)),
  utl_raw.compare(c_varbinary,c_raw),rawtohex(utl_raw.bit_and(c_varbinary,c_raw)),
  rawtohex(utl_raw.bit_or(c_varbinary,c_raw)),
  rawtohex(utl_raw.bit_xor(c_varbinary,c_raw)) from app.bin_t where id=2;
select utl_raw.cast_to_varchar2(c_varbinary),
  rawtohex(hextoraw('414243')),rawtohex(cast('414243' as varbinary(8)))
  from app.bin_t where id=1;
select min(c_binary),max(c_varbinary),count(c_raw),
  count(distinct c_raw) from app.bin_t;
select id from app.bin_t index idx_bin
  where c_varbinary>=0x00 and c_binary=0xAA order by c_varbinary;
create view app.bin_v as
  select id,c_binary,c_varbinary,c_raw,c_alter from app.bin_t
  where c_varbinary is not null;
select rawtohex(c_binary),rawtohex(c_varbinary),rawtohex(c_raw)
  from app.bin_v order by id;
update app.bin_v set c_raw=hextoraw('ABCD') where id=2;
create function app.bin_binary_f(p_value binary(32767)) return binary(32767) as
begin
  return p_value;
end;

create function app.bin_varbinary_f(p_value varbinary(32767)) return varbinary(32767) as
begin

end;

create function app.bin_raw_f(p_value raw(32767)) return raw(32767) as
begin
  return p_value;
end;
/

  utl_raw.length(app.bin_varbinary_f(0xABCD)),
  utl_raw.length(app.bin_raw_f(hextoraw('ABCDEF'))) from dual;
