DECLARE V XMLTYPE; BEGIN SELECT DOC INTO V FROM XMLTYPE_DOCS WHERE ID=1; PRINT V; END;
/
CREATE OR REPLACE PROCEDURE XMLTYPE_PROC(P_DOC IN XMLTYPE, P_RESULT OUT XMLTYPE) AS BEGIN P_RESULT=APPENDCHILDXML(P_DOC,'/ROOT','<P>1</P>'); END;
/
DECLARE V XMLTYPE; BEGIN XMLTYPE_PROC(XMLTYPE('<ROOT/>'),V); PRINT V; END;
/
