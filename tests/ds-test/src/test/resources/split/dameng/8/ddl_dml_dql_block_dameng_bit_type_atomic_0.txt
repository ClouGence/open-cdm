create table app.bit_t(
  id int primary key,flag bit default 0 not null,flag_null bit null,
  constraint ck_bit check(flag in(0,1)),constraint uk_bit unique(flag_null));
comment on column app.bit_t.flag is '0 false, nonzero true';
create index app.idx_bit on app.bit_t(flag,id);
alter table app.bit_t add flag_alter bit default 1;
insert into app.bit_t(id,flag) values(0,0);
insert into app.bit_t(id,flag) values(1,1);
insert into app.bit_t(id,flag) values(2,-1);
insert into app.bit_t(id,flag) values(3,2);
insert into app.bit_t(id,flag) values(4,0.0);
insert into app.bit_t(id,flag) values(5,0.1);
insert into app.bit_t(id,flag) values(6,'000');
insert into app.bit_t(id,flag) values(7,'10');
insert into app.bit_t(id) values(8);
update app.bit_t set flag=case when flag=0 then 1 else 0 end,
  flag_alter=case when flag_alter=0 then 1 else 0 end where id in(0,1);
select id,flag,flag_null,flag_alter,cast(flag as int)
  from app.bit_t order by id;
select sum(case when flag=0 then 1 else 0 end),
  sum(case when flag=1 then 1 else 0 end),min(flag),max(flag),
  sum(cast(flag as int)),avg(cast(flag as int)) from app.bit_t;
select id from app.bit_t index idx_bit where flag=1 order by id;
select bitand(cast(flag as int),cast(flag_alter as int)),
  mod(cast(flag as int),2),abs(cast(flag as int)-1),
  cast(flag as varchar(5)),cast(flag as boolean) from app.bit_t order by id;
select data_type,data_length,data_precision,data_scale
  from user_tab_columns where table_name='BIT_T' order by column_id;
create view app.bit_v as
  select id,flag,flag_alter from app.bit_t where flag=1;
select id,case when flag then 'T' else 'F' end
  from app.bit_v order by id;
update app.bit_v set flag_alter=0 where id=2;
create function app.bit_f(p bit) return bit as
begin return p;end;
/
select app.bit_f(0),app.bit_f(1),app.bit_f(-9223372036854775808),
  app.bit_f(9223372036854775807),app.bit_f(null) from dual;
insert into app.bit_t(id,flag) values(100,'1e1');
insert into app.bit_t(id,flag) values(100,'3.14');
insert into app.bit_t(id,flag) values(100,'2a5');
create table app.bad_bit_partition(id bit,v int)
  partition by list(id)(partition p0 values(0),partition p1 values(1));
create table app.bad_bit_arg_1(c bit(1));
create table app.bad_bit_arg_0(c bit(0));
insert into app.bit_t(id,flag) values(101,null);
drop view app.bit_v;
drop function app.bit_f;
drop table app.bit_t;
