create table app.exact_num_t(
  id int primary key,
  c_numeric numeric(38,0) default 0 not null,
  c_decimal decimal(38,38) default 0,
  c_dec dec(10,2) default 0 check(c_dec between -99999999.99 and 99999999.99),
  c_number number(10,2) default 0,
  c_number_default number,
  c_null numeric null,
  constraint uk_exact_num unique(c_numeric,c_decimal,c_dec,c_number)
);
create index app.idx_exact_num on app.exact_num_t(c_numeric,c_number,c_dec);

  from app.exact_num_t order by id;

  abs(c_numeric),power(c_dec,2) from app.exact_num_t where id=4;
select cast(c_numeric as varchar(40)),cast(c_decimal as double),
  cast(c_dec as int),cast(c_number as bigint),
  cast(c_number_default as decimal(38,1)) from app.exact_num_t where id=4;
select min(c_numeric),max(c_decimal),sum(c_dec),avg(c_number),
  count(c_number_default) from app.exact_num_t;
select id from app.exact_num_t index idx_exact_num
  where c_numeric>=0 and c_number>=0 order by c_numeric;
create view app.exact_num_v as
  select id,c_numeric,c_decimal,c_dec,c_number from app.exact_num_t where c_numeric>=0;
select c_numeric+c_dec as num_dec_sum,
  c_number*c_dec as number_dec_product from app.exact_num_v
  where id in(3,4) order by id;
update app.exact_num_v set c_number=c_number+0.01 where id=4;
