-- REJECT_004_JSON_TABLE_EXISTS_MISSING_PATH

-- REJECT_005_JSON_TABLE_FORMAT_MISSING_JSON

SELECT * FROM JSON_TABLE('{}', '$' COLUMNS(NESTED PATH '$.x'));

SELECT JSON_VALUE('{}', '$.x' ON EMPTY);
-- REJECT_008_JSON_QUERY_WRAPPER_MISSING_WITH
SELECT JSON_QUERY('{"a":1}', '$.a' ARRAY WRAPPER);
