create table app.int_type_t(
  id int primary key,
  c_tiny tinyint default 0 not null check(c_tiny between -128 and 127),
  c_byte byte default 0,
  c_small smallint default 0,
  c_integer integer default 0,
  c_int int default 0,
  c_big bigint default 0,
  c_null bigint null,
  constraint uk_int_types unique(c_tiny,c_byte,c_small,c_integer,c_int,c_big)
);
create index app.idx_int_types on app.int_type_t(c_big,c_integer,c_small,c_tiny);

update app.int_type_t set c_tiny=c_tiny+1,c_byte=c_byte+1,
  c_small=c_small+1,c_integer=c_integer+1,c_int=c_int+1,c_big=c_big+1 where id=4;
select id,c_tiny,c_byte,c_small,c_integer,c_int,c_big,c_null
  from app.int_type_t order by id;
select min(c_tiny),max(c_byte),sum(c_small),avg(c_integer),count(c_int),
  abs(min(c_big)+1) from app.int_type_t;
select cast(c_tiny as varchar(10)),cast(c_byte as int),
  cast(c_small as bigint),cast(c_integer as decimal(38,0)),cast(c_int as double),
  cast(c_big as varchar(30)) from app.int_type_t where id=4;
select id from app.int_type_t index idx_int_types
  where c_big>=0 and c_integer>=0 order by c_big;
create view app.int_type_v as
  select id,c_tiny,c_byte,c_small,c_integer,c_int,c_big
  from app.int_type_t where c_big>=0;
select c_tiny+c_byte as tiny_byte_sum,c_small*c_int as small_int_product,
  mod(c_big,2) as big_parity from app.int_type_v where id in(3,4) order by id;
update app.int_type_v set c_big=c_big+1 where id=4;
