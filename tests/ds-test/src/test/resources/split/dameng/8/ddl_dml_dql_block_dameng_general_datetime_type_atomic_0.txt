create table app.dt_t(
  id int primary key,c_date date default date '2000-01-01' not null,
  c_time0 time(0) default time '00:00:00',c_time6 time(6),
  c_ts0 timestamp(0),c_ts9 timestamp(9),c_ts timestamp,
  c_dt0 datetime(0),c_dt9 datetime(9),c_dt datetime,
  c_null timestamp null,
  constraint ck_dt check(c_date>=date '-4712-01-01'),
  constraint uk_dt unique(c_ts9));
comment on column app.dt_t.c_date is 'Gregorian and Julian date';
comment on column app.dt_t.c_ts9 is 'declared nanosecond timestamp';
create index app.idx_dt on app.dt_t(c_date,c_ts9);
