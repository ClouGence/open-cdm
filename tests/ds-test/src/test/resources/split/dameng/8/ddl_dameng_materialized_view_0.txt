create materialized view mv_basic as select id, customer_id, amount, status from mv_orders;
create materialized view mv_columns(order_id, order_amount) as select id, amount from mv_orders;
create materialized view mv_build_immediate build immediate as select id, amount from mv_orders;
create materialized view mv_build_deferred build deferred as select id, amount from mv_orders;
create materialized view mv_refresh_complete refresh complete as select id, amount from mv_orders;
create materialized view mv_refresh_force refresh force as select id, amount from mv_orders;
create materialized view mv_never_refresh never refresh as select id, amount from mv_orders;
create materialized view mv_on_demand refresh complete on demand as select id, amount from mv_orders;
create materialized view mv_primary_key refresh complete with primary key as select id, amount from mv_orders;
create materialized view mv_rowid refresh complete with rowid as select id, amount from mv_orders;
create materialized view mv_using_default refresh complete using default as select id, amount from mv_orders;
create materialized view mv_using_truncate refresh complete using truncate as select id, amount from mv_orders;
create materialized view mv_using_delete refresh complete using delete as select id, amount from mv_orders;
create materialized view mv_disable_rewrite disable query rewrite as select id, amount from mv_orders;
create materialized view mv_enable_rewrite enable query rewrite as select id, amount from mv_orders;
create materialized view mv_filtered as select id, amount from mv_orders where status = 'OPEN';
create materialized view mv_join as select o.id, o.amount, c.name, c.region from mv_orders o join mv_customers c on o.customer_id = c.id;
create materialized view mv_aggregate as select customer_id, count(*) as order_count, sum(amount) as total_amount from mv_orders group by customer_id;
create materialized view mv_subquery as select id, amount from mv_orders where customer_id in(select id from mv_customers where region = 'EAST');
create materialized view mv_union as select id, amount from mv_orders where status = 'OPEN' union all select id, amount from mv_orders where status = 'CLOSED';
create materialized view mv_ordered as select id, amount from mv_orders order by id;
create materialized view mv_distinct as select distinct customer_id from mv_orders;
create materialized view mv_expression as select id, amount * 1.1 as adjusted_amount from mv_orders;
create materialized view mv_case as select id, case when amount >= 100 then 'HIGH' else 'LOW' end as amount_level from mv_orders;
create materialized view mv_cte as with open_orders as (select id, amount from mv_orders where status = 'OPEN') select * from open_orders;
