-- CASE_001_JSON_CONCAT_OBJECT
SELECT JSON_CONCAT('{"a":1}', '{"b":2}');
-- CASE_002_JSON_CONCAT_ARRAY
SELECT JSON_CONCAT('[1,2]', '[3,4]');
-- CASE_003_JSON_CONCAT_SCALAR
SELECT JSON_CONCAT('1', '"x"');
-- CASE_004_JSON_CONCAT_NULL
SELECT JSON_CONCAT(NULL, '{"a":1}');
-- CASE_005_JSON_VALID_VALID
SELECT JSON_VALID('{"a":[1,true,null]}');
-- CASE_006_JSON_VALID_INVALID
SELECT JSON_VALID('{"a":');
-- CASE_007_JSON_VALID_NULL
SELECT JSON_VALID(NULL);
-- CASE_008_ROW_TO_JSON_TABLE
SELECT ROW_TO_JSON(T) FROM app.json_docs T ORDER BY ID;
-- CASE_009_ROW_TO_JSON_ROW
SELECT ROW_TO_JSON(ROW(ID, PAYLOAD))
FROM app.json_docs ORDER BY ID;
-- CASE_010_ROW_TO_JSON_ALIAS
SELECT ROW_TO_JSON(Q)
FROM (SELECT ID, PAYLOAD FROM app.json_docs WHERE ID=1) Q;
-- CASE_011_JSON_TABLE_EXISTS
SELECT * FROM JSON_TABLE(
  '{a:100,b:200,c:{d:300,e:400}}', '$.c'
  COLUMNS(C1 INT EXISTS PATH '$.d' FALSE ON EMPTY,
          C2 INT EXISTS PATH '$.f' FALSE ON EMPTY)
);
-- CASE_012_JSON_TABLE_FORMAT_JSON
SELECT * FROM JSON_TABLE(
  '[{a:100,c:{d:300}},{a:500,c:{d:700}}]', '$[*]'
  COLUMNS(C1 VARCHAR2(20) FORMAT JSON PATH '$.c')
);
-- CASE_013_JSON_TABLE_NESTED
SELECT * FROM JSON_TABLE(
  '{"a":100,"c":{"d":300,"e":400}}', '$'
  COLUMNS(NESTED PATH '$.c' COLUMNS(C1 INT PATH '$.d', C2 INT PATH '$.e'))
);
-- CASE_014_JSON_TABLE_ORDINALITY_FORMAT
SELECT * FROM JSON_TABLE(
  '[{a:100,c:{d:300}},{a:500,c:{d:700}}]', '$[*]'
  COLUMNS(ID FOR ORDINALITY, C1 VARCHAR2(20) FORMAT JSON PATH '$.c')
);
-- CASE_015_JSON_TABLE_VALUE_EMPTY_DEFAULT
SELECT * FROM JSON_TABLE(
  '{"a":1}', '$'
  COLUMNS(MISSING INT PATH '$.missing' DEFAULT '9' ON EMPTY)
);
-- CASE_016_JSON_TABLE_QUERY_EMPTY_ARRAY
SELECT * FROM JSON_TABLE(
  '{"a":1}', '$'
  COLUMNS(MISSING VARCHAR2(20) FORMAT JSON PATH '$.missing' EMPTY ARRAY ON EMPTY)
);
-- CASE_017_JSON_VALUE_EMPTY_DEFAULT
SELECT JSON_VALUE('{"a":1}', '$.missing' RETURNING NUMBER DEFAULT '9' ON EMPTY);
-- CASE_018_JSON_VALUE_EMPTY_NULL
SELECT JSON_VALUE('{"a":1}', '$.missing' RETURNING NUMBER NULL ON EMPTY);
-- SEM_001_JSON_OBJECT_ODD_ARGUMENTS
SELECT JSON_OBJECT('a', 1, 'b');
-- SEM_002_JSON_OBJECT_KEYS_NOT_FROM
SELECT JSON_OBJECT_KEYS('{"a":1}');
-- SEM_003_JSON_TYPEOF_INVALID
SELECT JSON_TYPEOF('not-json');
-- SEM_004_JSONB_EACH_NON_OBJECT
SELECT * FROM JSONB_EACH('[1,2]');
-- SEM_005_JSON_TABLE_ERROR_ON_EMPTY
SELECT * FROM JSON_TABLE(
  '{"a":1}', '$' COLUMNS(MISSING INT PATH '$.missing' ERROR ON EMPTY)
);
-- SEM_006_JSON_TABLE_FORMAT_SCALAR
SELECT * FROM JSON_TABLE(
  '{"a":1}', '$' COLUMNS(A VARCHAR2(20) FORMAT JSON PATH '$.a' ERROR ON ERROR)
);
-- SEM_007_JSON_CONCAT_MISSING_SECOND
SELECT JSON_CONCAT('{"a":1}');
-- SEM_008_JSON_VALID_EXTRA_ARGUMENT
SELECT JSON_VALID('{}', '{}');
-- SEM_009_ROW_TO_JSON_EMPTY
SELECT ROW_TO_JSON();
-- SEM_010_JSONB_SET_MISSING_VALUE
SELECT JSONB_SET('{}', '{a}');
