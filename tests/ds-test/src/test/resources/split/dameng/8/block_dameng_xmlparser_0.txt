declare
  parser_handle dbms_xmlparser.parser;
  document_node dbms_xmldom.domdocument;
  root_node dbms_xmldom.domnode;
  child_nodes dbms_xmldom.domnodelist;

  parser_handle := dbms_xmlparser.newparser();
  dbms_xmlparser.parsebuffer(parser_handle, '<ROOT><A/><B/><C/></ROOT>');

  root_node := dbms_xmldom.makenode(dbms_xmldom.getdocumentelement(document_node));

  dbms_xmldom.freenodelist(child_nodes);
  dbms_xmlparser.freeparser(parser_handle);
end;
