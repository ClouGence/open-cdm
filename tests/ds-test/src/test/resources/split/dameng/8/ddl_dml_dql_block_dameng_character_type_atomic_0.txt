create table app.char_t(
  id int primary key,c_char char(4) default 'A' not null,
  c_character character(4) default 'B',c_varchar varchar(12) default 'C',
  c_varchar2 varchar2(12) default 'D',c_char_sem char(4 char),
  c_character_sem character(4 char),c_varchar_sem varchar(4 char),
  c_varchar2_sem varchar2(4 char),c_char_byte char(12 byte),
  c_character_byte character(12 byte),c_varchar_byte varchar(12 byte),
  c_varchar2_byte varchar2(12 byte),c_null varchar2(12) null,
  constraint ck_char check(length(c_varchar)<=12),
  constraint uk_char unique(c_varchar,c_varchar2));
comment on column app.char_t.c_char is 'fixed-width character';
comment on column app.char_t.c_varchar2 is 'variable-width character';
create index app.idx_char on app.char_t(c_varchar,c_char);
alter table app.char_t add c_alter varchar2(8) default 'ALTERED';
alter table app.char_t modify c_alter varchar2(16);
insert into app.char_t(id,c_char,c_character,c_varchar,c_varchar2,
  c_char_sem,c_character_sem,c_varchar_sem,c_varchar2_sem,c_char_byte,
  c_character_byte,c_varchar_byte,c_varchar2_byte,c_null)
values(1,'A','B','alpha','Beta','达梦数据','达梦数据','达梦数据','达梦数据',
  '达梦数据','达梦数据','达梦数据','达梦数据',null);
insert into app.char_t(id,c_char,c_character,c_varchar,c_varchar2,
  c_char_sem,c_character_sem,c_varchar_sem,c_varchar2_sem,c_null)
values(2,'XY','ZW','gamma','delta','AB','CD','EF','GH','nullable');
insert into app.char_t(id) values(3);
update app.char_t set c_varchar=upper(c_varchar),
  c_varchar2=lower(c_varchar2),c_alter=c_alter||'_OK' where id=1;
select id,length(c_char),lengthb(c_char),length(c_character),
  lengthb(c_character),length(c_varchar),lengthb(c_varchar),
  length(c_varchar2),lengthb(c_varchar2) from app.char_t order by id;
select trim(c_char),ltrim(c_character),rtrim(c_character),
  substr(c_varchar,1,3),instr(c_varchar2,'t'),replace(c_varchar,'A','a')
  from app.char_t where id=1;
select concat(c_varchar,c_varchar2),c_char||c_character,
  lpad(c_varchar,12,'_'),rpad(c_varchar2,12,'_'),ascii(c_char),chr(65)
  from app.char_t where id=1;
select lower(c_varchar),upper(c_varchar2),initcap(c_varchar),
  reverse(c_varchar2),translate(c_varchar,'abc','ABC')
  from app.char_t where id=1;
select cast(c_char as varchar2(20)),cast(c_varchar as char(12)),
  convert(c_varchar2,'UTF-8','UTF-8'),to_char(12345),to_number('12345')
  from app.char_t where id=1;
select min(c_char),max(c_character),count(c_varchar),
  count(distinct c_varchar2) from app.char_t;
select id from app.char_t index idx_char
  where c_varchar>='A' and c_char='A' order by c_varchar;
create view app.char_v as
  select id,c_char,c_character,c_varchar,c_varchar2,c_alter
  from app.char_t where c_varchar is not null;
select c_char||c_character as fixed_pair,
  c_varchar||':'||c_varchar2 as variable_pair from app.char_v order by id;
update app.char_v set c_varchar2='viewedit' where id=2;
