-- UNION missing right branch
select 1 from dual union;
-- INTERSECT missing left branch
intersect select 1 from dual;
-- duplicate quantifier
select 1 from dual union all distinct select 2 from dual;
-- CORRESPONDING missing branch
select 1 a from dual union corresponding by(a);

select 1 a from dual union corresponding by() select 2 a from dual;

select 1 from dual minus;
