  FILE_VALUE BFILE;
  DIRECTORY_NAME VARCHAR(128);
  FILE_NAME VARCHAR(256);
BEGIN

  DBMS_LOB.FILEGETNAME(FILE_VALUE, DIRECTORY_NAME, FILE_NAME);
END;
