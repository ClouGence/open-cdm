declare
  b0 bool:=false;b1 boolean:=true;bn boolean:=null;
  i0 int;i1 int;ineg int;bnot bool;
begin
  b0:=0;b1:=-7;i0:=cast(b0 as int);i1:=cast(b1 as int);
  bnot:=not b0;ineg:=cast(bnot as int);
  if b1 and not b0 or bn is null then
    dbms_output.put_line('BOOL='||i0||','||i1||','||ineg);
  end if;
end;

  type bool_record is record(a bool,b boolean);
  type bool_array is array boolean[3];

  dbms_output.put_line(cast(t(1) as int)||','||cast(t(2) as int));
end;

  return boolean as
