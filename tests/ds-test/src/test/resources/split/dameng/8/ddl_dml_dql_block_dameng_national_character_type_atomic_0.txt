create table app.nchar_t(
  id int primary key,c_nchar nchar(4) default 'A' not null,
  c_nvarchar nvarchar(4) default 'B',c_nvarchar2 nvarchar2(4) default 'C',
  c_national_char national char(4) default 'D',
  c_national_character national character(4) default 'E',
  c_nchar_varying nchar varying(4) default 'F',
  c_national_varying national character varying(4) default 'G',
  c_null nvarchar2(4) null,
  constraint ck_nchar check(length(c_nvarchar)<=4),
  constraint uk_nchar unique(c_nvarchar,c_nvarchar2))
storage(using long row);
comment on column app.nchar_t.c_nchar is 'national fixed character';
comment on column app.nchar_t.c_national_varying
  is 'national varying character';
create index app.idx_nchar on app.nchar_t(c_nvarchar,c_nchar);
alter table app.nchar_t add c_alter nvarchar2(8) default 'ALTERED';
alter table app.nchar_t modify c_alter nvarchar2(16);
insert into app.nchar_t values(
  1,'AB','CD','EF','GH','IJ','KL','MN',null,'ALTERED');
insert into app.nchar_t(id) values(2);
update app.nchar_t set c_nvarchar=upper(c_nvarchar),
  c_nvarchar2=lower(c_nvarchar2),c_alter=c_alter||'_OK' where id=1;
select id,length(c_nchar),lengthb(c_nchar),length(c_nvarchar),
  lengthb(c_nvarchar),length(c_national_char),lengthb(c_national_char),
  length(c_nchar_varying),lengthb(c_nchar_varying)
  from app.nchar_t order by id;
select trim(c_nchar),substr(c_nvarchar,1,2),instr(c_nvarchar2,'F'),
  replace(c_national_char,'G','g'),concat(c_nchar_varying,c_national_varying),
  upper(c_national_character),lower(c_nvarchar2)
  from app.nchar_t where id=1;
select cast(c_nchar as varchar2(20)),cast(c_nvarchar as nchar(8)),
  min(c_national_char),max(c_national_character),count(c_nchar_varying)
  from app.nchar_t group by c_nchar,c_nvarchar;
select id from app.nchar_t index idx_nchar
  where c_nvarchar>='A' order by c_nvarchar;
create view app.nchar_v as
  select id,c_nchar,c_nvarchar,c_nvarchar2,c_national_char,
  c_national_character,c_nchar_varying,c_national_varying
  from app.nchar_t where c_nvarchar is not null;
select c_nchar||c_nvarchar,
  c_national_char||c_national_character,c_nchar_varying||c_national_varying
  from app.nchar_v order by id;
update app.nchar_v set c_nvarchar2='VIEW' where id=1;
