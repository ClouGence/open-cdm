create table trg_target(id int primary key, name varchar(30), amount int);
create table trg_log(seq int identity, trigger_name varchar(40), operation varchar(10), old_id int, new_id int);
create view trg_view as select id, name, amount from trg_target;
create or replace trigger trg_before_insert
before insert on trg_target
referencing new row as n
for each row
begin
  :n.name = upper(:n.name);
end;
/
create or replace trigger trg_after_insert
after insert on trg_target
referencing new as n
for each row
begin
  insert into trg_log(trigger_name, operation, new_id)
  values('TRG_AFTER_INSERT', 'INSERT', :n.id);
end;
/
insert into trg_target values(1, 'alpha', 10);
select id, name, amount from trg_target where id = 1;
select trigger_name, operation, new_id from trg_log where new_id = 1;
create or replace trigger trg_before_update
before update of name, amount on trg_target
referencing old row as o new row as n
for each row
when (:n.amount > 0)
begin
  :n.amount = :o.amount + :n.amount;
end;
/
update trg_target set name = 'beta', amount = 5 where id = 1;
select id, name, amount from trg_target where id = 1;
create or replace trigger trg_after_delete
after delete on trg_target
referencing old as o
for each row
begin
  insert into trg_log(trigger_name, operation, old_id)
  values('TRG_AFTER_DELETE', 'DELETE', :o.id);
end;
/
delete from trg_target where id = 1;
select trigger_name, operation, old_id from trg_log where old_id = 1;
create or replace trigger trg_all_dml
after insert or delete or update on trg_target
for each row
declare
  v_operation varchar(10);
  v_old_id int;
  v_new_id int;
begin
  if inserting then
    v_operation = 'INSERT';
    v_new_id = :new.id;
  elseif updating then
    v_operation = 'UPDATE';
    v_old_id = :old.id;
    v_new_id = :new.id;
  else
    v_operation = 'DELETE';
    v_old_id = :old.id;
  end if;
  insert into trg_log(trigger_name, operation, old_id, new_id)
  values('TRG_ALL_DML', v_operation, v_old_id, v_new_id);
end;
/
insert into trg_target values(2, 'gamma', 20);
update trg_target set amount = amount + 1 where id = 2;
delete from trg_target where id = 2;
select operation, old_id, new_id from trg_log where trigger_name = 'TRG_ALL_DML' order by seq;
create or replace trigger trg_before_update_statement
before update on trg_target
for each statement
begin
  insert into trg_log(trigger_name, operation) values('TRG_BEFORE_STMT', 'UPDATE');
end;
/
create or replace trigger trg_after_update_statement
after update on trg_target
begin
  insert into trg_log(trigger_name, operation) values('TRG_AFTER_STMT', 'UPDATE');
end;
/
insert into trg_target values(3, 'one', 1);
insert into trg_target values(4, 'two', 2);
update trg_target set amount = amount + 10 where id in (3, 4);
select trigger_name, count(*) from trg_log where trigger_name in ('TRG_BEFORE_STMT', 'TRG_AFTER_STMT') group by trigger_name order by trigger_name;
create or replace trigger trg_view_insert
instead of insert on trg_view
for each row
begin
  insert into trg_target values(:new.id, :new.name, :new.amount);
end;
/
insert into trg_view values(5, 'view_insert', 50);
select id, name, amount from trg_target where id = 5;
create or replace trigger trg_view_update
instead of update on trg_view
for each row
begin
  update trg_target set name = :new.name, amount = :new.amount where id = :old.id;
end;
/
update trg_view set name = 'view_update', amount = 55 where id = 5;
select id, name, amount from trg_target where id = 5;
create or replace trigger trg_view_delete
instead of delete on trg_view
for each row
begin
  delete from trg_target where id = :old.id;
end;
/
delete from trg_view where id = 5;
select count(*) from trg_target where id = 5;

before insert on trg_target
for each row
begin
  insert into trg_log(trigger_name, operation, new_id) values('ORDER_FIRST', 'INSERT', :new.id);
end;
/

before insert on trg_target
for each row
follows trg_order_first
begin
  insert into trg_log(trigger_name, operation, new_id) values('ORDER_SECOND', 'INSERT', :new.id);
end;
/
create or replace trigger trg_order_zero
before insert on trg_target
for each row
precedes trg_order_first
begin
  insert into trg_log(trigger_name, operation, new_id) values('ORDER_ZERO', 'INSERT', :new.id);
end;
/
