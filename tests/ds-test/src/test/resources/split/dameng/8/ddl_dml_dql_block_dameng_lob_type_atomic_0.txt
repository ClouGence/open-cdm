  lengthb(c_image),lengthb(c_longvarbinary),lengthb(c_blob),length(c_clob)
  from app.lob_t order by id;

  from app.lob_t where id=1;
select case when c_text='ABCD-UPDATE' then 1 else 0 end,
  case when c_clob='达梦CLOB-UPDATE' then 1 else 0 end from app.lob_t where id=1;
select c_text from app.lob_t where c_text=(select c_clob from app.lob_t where id=3);
select c_text,count(*) from app.lob_t group by c_text;
select c_clob from app.lob_t order by c_clob;
select distinct c_blob from app.lob_t;
select min(c_text),max(c_clob),min(c_blob) from app.lob_t;
create view app.lob_v as select id,c_text,c_image,c_blob,c_clob
  from app.lob_t where id>=1;
select id,length(c_text),lengthb(c_blob),length(c_clob) from app.lob_v order by id;
