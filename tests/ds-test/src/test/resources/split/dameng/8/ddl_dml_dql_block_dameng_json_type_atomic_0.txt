create table app.json_types(
 id int primary key,j json default '{}',jb jsonb default '{}' not null,
 v varchar2(32767) check(v is json(strict) with unique keys),
 c clob check(c is json(lax) without unique keys),
 t text check(t is json),l longvarchar check(l is json)) storage(using long row);
comment on table app.json_types is 'JSON type matrix';
comment on column app.json_types.j is 'text preserving JSON';
comment on column app.json_types.jb is 'normalized JSONB';
comment on column app.json_types.v is 'strict unique JSON';
comment on column app.json_types.c is 'lax nonunique JSON';
comment on column app.json_types.t is 'TEXT JSON';
