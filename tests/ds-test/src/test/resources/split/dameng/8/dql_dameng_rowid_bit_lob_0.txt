SELECT ID, RID, ROWID AS PHYSICAL_ROWID FROM CODEX_DM8_TYPE_ROWID ORDER BY RID;
SELECT MIN(RID) AS MIN_RID, MAX(RID) AS MAX_RID FROM CODEX_DM8_TYPE_ROWID;
SELECT ID, FLAG, CAST(FLAG AS INT) AS FLAG_INT FROM CODEX_DM8_TYPE_BIT ORDER BY ID;
SELECT COUNT(*) AS TRUE_COUNT FROM CODEX_DM8_TYPE_BIT WHERE FLAG = 1;
SELECT COUNT(*) AS FALSE_COUNT FROM CODEX_DM8_TYPE_BIT WHERE FLAG = 0;
SELECT ID, LENGTH(V) AS VALUE_LENGTH FROM CODEX_DM8_TYPE_TEXT;
SELECT ID, LENGTH(V) AS VALUE_LENGTH FROM CODEX_DM8_TYPE_LONG;
SELECT ID, LENGTH(V) AS VALUE_LENGTH FROM CODEX_DM8_TYPE_LONGVARCHAR;
SELECT ID, LENGTHB(V) AS VALUE_LENGTH FROM CODEX_DM8_TYPE_IMAGE;
SELECT ID, LENGTHB(V) AS VALUE_LENGTH FROM CODEX_DM8_TYPE_LONGVARBINARY;
SELECT ID, DBMS_LOB.GETLENGTH(V) AS VALUE_LENGTH FROM CODEX_DM8_TYPE_BLOB;
SELECT ID, DBMS_LOB.GETLENGTH(V) AS VALUE_LENGTH FROM CODEX_DM8_TYPE_CLOB;
SELECT ID, DBMS_LOB.FILEEXISTS(V) AS FILE_EXISTS, DBMS_LOB.GETLENGTH(V) AS FILE_LENGTH
FROM CODEX_DM8_TYPE_BFILE;
SELECT TEXT_EQUAL(V, 'TEXT-VALUE') AS IS_EQUAL FROM CODEX_DM8_TYPE_TEXT;
SELECT BLOB_EQUAL(V, 0x494D414745) AS IS_EQUAL FROM CODEX_DM8_TYPE_IMAGE;
SELECT DBMS_LOB.FILEEXISTS(BFILENAME('CODEX_DM8_BFILE_DIR', 'missing.bin')) FROM DUAL;
SELECT TABLE_NAME, COLUMN_NAME, DATA_TYPE, DATA_LENGTH, DATA_PRECISION, DATA_SCALE
FROM USER_TAB_COLUMNS
WHERE TABLE_NAME LIKE 'CODEX_DM8_BAD_%' AND COLUMN_NAME = 'V'
ORDER BY TABLE_NAME;
