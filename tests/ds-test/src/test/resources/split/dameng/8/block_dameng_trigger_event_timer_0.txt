create table event_log(
  seq int identity,
  trigger_name varchar(40),
  operation varchar(20),
  object_type varchar(40),
  object_name varchar(128),
  schema_name varchar(128)
);
create or replace trigger evt_after_create
after create on CODEX_DM8_EVENT_TRIGGER_20260721.schema
begin
  insert into event_log(trigger_name, operation, object_type, object_name, schema_name)
  values('EVT_AFTER_CREATE', :eventinfo.optype, :eventinfo.objecttype,
         :eventinfo.objectname, :eventinfo.schemaname);
end;
/
create table event_table(id int, name varchar(20));
create view event_view as select id, name from event_table;
create sequence event_sequence start with 1 increment by 1;
select operation, object_type, object_name from event_log where trigger_name = 'EVT_AFTER_CREATE' order by seq;

create table event_sql_text(
  fragment_count int,
  sql_text varchar,
  object_name varchar(128),
  object_type varchar(128),
  object_owner varchar(128)
);
create or replace trigger evt_sql_text
before create on CODEX_DM8_EVENT_TRIGGER_20260721.schema
declare
  v_count int;
  v_text varchar;
  v_parts dm_name_list_t;
begin
  v_count = dm_sql_txt(v_parts);
  for i in 1..v_count loop
    v_text = v_text || v_parts(i);
  end loop;
  insert into event_sql_text values(v_count, v_text, dm_dict_obj_name,
    dm_dict_obj_type, dm_dict_obj_owner);
end;
/
create table event_text_target(id int);
select fragment_count, object_name, object_type, object_owner from event_sql_text where object_name = 'EVENT_TEXT_TARGET';
