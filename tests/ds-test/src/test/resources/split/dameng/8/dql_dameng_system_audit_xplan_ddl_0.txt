select para_name, para_value
from v$dm_ini
where para_name in ('COMPATIBLE_MODE', 'TRC_LOG', 'TRC_LOG_MODULE', 'USE_PLN_POOL', 'ENABLE_MONITOR_PLNHIST')
order by para_name;
select object_name, object_type, status
from dba_objects
where object_name in ('DBMS_SYSTEM', 'DBMS_XPLAN', 'DBMS_DDL', 'DBMS_SQL')
order by object_name, object_type;
select length(dbms_ddl.wrap(
  'CREATE OR REPLACE FUNCTION CODEX_DM8_WRAP_ONLY RETURN INT AS BEGIN RETURN 7; END;'
)) wrapped_length
from dual;
select substr(dbms_ddl.wrap(
  'CREATE OR REPLACE FUNCTION CODEX_DM8_WRAP_ONLY RETURN INT AS BEGIN RETURN 7; END;'
), 1, 34) wrapped_prefix
from dual;
select CODEX_DM8_WRAPPED_F1() wrapped_result from dual;
select name, type, substr(text, 1, 34) wrapped_source_prefix
from user_source
where name='CODEX_DM8_WRAPPED_F1'
order by line;
select CODEX_DM8_WRAPPED_F2() wrapped_array_result from dual;
select para_name, para_value
from v$dm_ini
where para_name='ENABLE_AUDIT';
select count(*) audit_file_count from v$audit_file;
select count(*) loaded_file_count from v$dbmsaudit_file;
select count(*) audit_record_count from v$auditrecords;
select count(*) selected_file_count from v$dbmsaudit_file;
select count(*) filtered_record_count from v$auditrecords;
