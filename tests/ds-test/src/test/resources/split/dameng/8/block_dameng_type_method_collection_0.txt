create or replace type t_methods as object(
  value int,
  static function add_two(a int, b int) return int,
  member function get_value return int,
  member procedure set_value(p_value int),
  map member function map_value return int
) not final;
/
create or replace type body t_methods as
  static function add_two(a int, b int) return int is
  begin
    return a + b;
  end;
  member function get_value return int is
  begin
    return value;
  end;
  member procedure set_value(p_value int) is
  begin
    value = p_value;
  end;
  map member function map_value return int is
  begin
    return value;
  end;
end;
/
select t_methods.add_two(2, 5);
select t_methods(7).get_value();
create table method_table(id int primary key, obj t_methods);
insert into method_table values(1, t_methods(8));
insert into method_table values(2, t_methods(3));
select id, t.obj.get_value(), t.obj.map_value() from method_table t order by id;
select id from method_table where obj > t_methods(4);
declare
  v_obj t_methods;
begin
  v_obj = new t_methods(1);
  v_obj.set_value(9);
  print v_obj.get_value();
end;

    return p_value * 2;
