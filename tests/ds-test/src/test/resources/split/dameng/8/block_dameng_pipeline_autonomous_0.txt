create type pipe_row_type as object(
  id int,
  name varchar(80)
);
/
create type pipe_table_type as table of pipe_row_type;
/
create or replace function pipe_all_rows return pipe_table_type pipelined
is
  v_row pipe_row_type;
begin
  for rec in (select id, name from pipeline_source order by id) loop
    v_row := pipe_row_type(rec.id, rec.name);
    pipe row(v_row);
  end loop;
exception
  when others then null;
end;
/
select * from table(pipe_all_rows()) order by id;
select p.id, p.name from table(pipe_all_rows()) p where p.id >= 2 order by p.id;
select count(*), min(id), max(id) from table(pipe_all_rows());
select p.id, s.name from table(pipe_all_rows()) p join pipeline_source s on s.id = p.id order by p.id;
with piped as (select * from table(pipe_all_rows())) select * from piped where id = 1;
select * from pipeline_source where id in (select id from table(pipe_all_rows())) order by id;
create or replace function pipe_limited_rows(p_limit int) return pipe_table_type authid current_user pipelined
is
begin
  for rec in (select id, name from pipeline_source where id <= p_limit order by id) loop
    pipe row(pipe_row_type(rec.id, rec.name));
  end loop;
end;
/
select * from table(pipe_limited_rows(2)) order by id;
select id, row_number() over(order by id) as rn from table(pipe_limited_rows(3));
create or replace function pipe_empty_rows return pipe_table_type deterministic pipelined
is
begin
  null;
exception
  when others then null;
end;
/
select count(*) from table(pipe_empty_rows());
