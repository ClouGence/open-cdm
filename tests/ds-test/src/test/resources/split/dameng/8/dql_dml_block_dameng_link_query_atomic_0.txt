  CURSOR C_REMOTE IS
    SELECT ID, NAME
    FROM remote_app.T_REMOTE@employee_link
    FOR UPDATE;
BEGIN
  FOR R IN C_REMOTE LOOP
    UPDATE remote_app.T_REMOTE@employee_link
    SET NAME=NAME WHERE CURRENT OF C_REMOTE;
  END LOOP;
END;
