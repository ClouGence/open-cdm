create table MPP_HASH(ID int primary key, GRP_ID int, NOTE varchar(40)) distributed by hash(ID);
create table MPP_RANDOM(ID int, NOTE varchar(40)) distributed randomly;
create table MPP_FULLY(ID int primary key, NOTE varchar(40)) distributed fully;
create table MPP_RANGE(ID int, NOTE varchar(40)) distributed by range(ID)
(values equ or less than(100) on EP01, values less than(maxvalue) on EP02);
create table MPP_LIST(ID int, NOTE varchar(40)) distributed by list(ID)
(values(1, 3, 5) on EP01, values(2, 4, 6) on EP02);
create table MPP_HASH_MULTI(ID int, GRP_ID int, NOTE varchar(40)) distributed by hash(ID, GRP_ID);
create table MPP_CTAS_HASH as select 1 as ID, 'A' as NOTE from DUAL distributed by hash(ID);
create table MPP_CTAS_RANDOM as select 2 as ID, 'B' as NOTE from DUAL distributed randomly;
create table MPP_HASH_PART(ID int, NOTE varchar(40))
partition by range(ID)(partition P1 values less than(10), partition P2 values less than(maxvalue))
distributed by hash(ID);
truncate table MPP_RANDOM;
