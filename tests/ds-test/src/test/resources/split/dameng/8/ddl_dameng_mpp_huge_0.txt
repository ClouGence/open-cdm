create huge table mpp_huge_hash(id int, note varchar(40)) storage(stat none, with delta, section(1024), filesize(16)) distributed by hash(id);
create huge table mpp_huge_random(id int, note varchar(40)) storage(stat none, with delta, section(1024), filesize(16)) distributed randomly;
create huge table mpp_huge_fully(id int, note varchar(40)) storage(stat none, with delta, section(1024), filesize(16)) distributed fully;
create huge table mpp_huge_range(id int, note varchar(40)) storage(stat none, with delta, section(1024), filesize(16)) distributed by range(id)(values equ or less than(100) on ep01, values less than(maxvalue) on ep02);
create huge table mpp_huge_list(id int, note varchar(40)) storage(stat none, with delta, section(1024), filesize(16)) distributed by list(id)(values(1,3,5) on ep01, values(2,4,6) on ep02);
create huge table mpp_huge_part(id int, note varchar(40)) partition by range(id)(partition p10 values less than(10), partition pmax values less than(maxvalue)) storage(stat none, with delta, section(1024), filesize(16)) distributed by hash(id);
create huge table mpp_huge_ctas as select 8 as id, 'CTAS' as note from dual distributed by hash(id);
create huge table mpp_huge_random_pk(id int primary key) distributed randomly;
create huge table mpp_huge_fully_pk(id int primary key) distributed fully;
