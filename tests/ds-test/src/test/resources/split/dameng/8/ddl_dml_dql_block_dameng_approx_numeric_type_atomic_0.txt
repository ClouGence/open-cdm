create table app.approx_t(
  id int primary key,c_float float(126) default 0 not null,c_float1 float(1),
  c_float24 float(24),c_float25 float(25),c_double double(126) default 0,
  c_real real default 0,c_dp double precision(126) default 0,c_null double null,
  constraint uk_approx unique(c_float,c_double,c_real,c_dp));
create index app.idx_approx on app.approx_t(c_double,c_float,c_real);

select id,c_float,c_float1,c_float24,c_float25,c_double,c_real,c_dp,c_null
  from app.approx_t order by id;
select round(c_float,2),trunc(c_double,2),abs(c_real),sqrt(c_dp),
  power(c_float,2),exp(c_real),ln(c_dp),sin(c_float),cos(c_double)
  from app.approx_t where id=4;
select cast(c_float as varchar(40)),cast(c_double as decimal(20,4)),
  cast(c_real as int),cast(c_dp as bigint) from app.approx_t where id=4;
select min(c_float),max(c_double),sum(c_real),avg(c_dp),count(c_double)
  from app.approx_t;
select id from app.approx_t index idx_approx
  where c_double>=0 and c_float>=0 order by c_double;
create view app.approx_v as
  select id,c_float,c_double,c_real,c_dp from app.approx_t where c_double>=0;
select c_float+c_real as float_real_sum,c_double*c_dp as double_product
  from app.approx_v where id in(3,4) order by id;
update app.approx_v set c_double=c_double+0.5 where id=4;
