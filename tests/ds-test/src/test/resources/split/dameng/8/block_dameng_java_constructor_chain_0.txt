create or replace java class j_this_chain
{
  int value;
  varchar name;
  int code;
  j_this_chain()
  {
    this.value = 1;
    this.name = 'ZERO';
    this.code = 0;
  }
  j_this_chain(int p_value)
  {
    this();
    this.value = p_value;
    this.name = 'ONE';
  }
  j_this_chain(int p_value, varchar p_name)
  {
    this(p_value);
    this.name = p_name;
    this.code = 2;
  }
  public int get_value()
  {
    return value;
  }
  public varchar get_name()
  {
    return name;
  }
  public int get_code()
  {
    return code;
  }
}
