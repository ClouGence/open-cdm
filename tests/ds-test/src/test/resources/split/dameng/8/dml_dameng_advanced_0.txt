update dml_target_a a set a.name = s.name from dml_source s where a.id = s.id + 9;
update dml_target_a a, dml_source s set a.amount = s.amount where a.id = s.id + 9;
update (select id, name, amount from dml_target_a where id = 10) t set t.name = 'SUBQUERY_TARGET';
update dml_partition_target partition(p_low) set name = upper(name) where id = 14;
update dml_partition_target partition for(15) set name = lower(name) where id = 15;
update dml_target_a index idx_dml_target_a_id set amount = amount + 1 where id = 13;
update dml_target_a set status = case category when 1 then 'A' when 2 then 'B' else 'C' end where id is not null;
update dml_target_a set amount = default where id = 10;
update dml_target_a set name = coalesce(name, 'UNKNOWN'), amount = nvl(amount, 0) where id is null;
update dml_target_a set amount = amount + 5 where id in(select id + 9 from dml_source);
update dml_target_a set status = 'Y' where exists(select 1 from dml_source s where s.id + 9 = dml_target_a.id);
with s as(select id + 9 id from dml_source)
update dml_target_a set status = 'C' where id in(select id from s);
delete from dml_target_a where id is null;

delete from dml_target_a a where a.id = -2;
delete from dml_partition_target partition(p_low) where id = -3;
delete from dml_partition_target partition for(15) where id = -4;
delete from (select id, name from dml_target_a where id < 0) t where t.id = -5;
with s as(select -6 id from dual)
delete from dml_target_a where id in(select id from s);
delete from dml_target_a where exists(select 1 from dml_source where dml_source.id = -7);
delete from dml_target_a where id = any(select id from dml_source where id < 0);
delete from dml_target_a where id not in(select id + 9 from dml_source) and id < 0;
