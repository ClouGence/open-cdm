create or replace java class j_parent
{
  int base_value;
  public int get_value()
  {
    return base_value;
  }
}
/
create or replace java class j_child extends j_parent
{
  int child_value;
  public override int get_value()
  {
    return child_value;
  }
}
/
select j_child(10, 20).get_value();
declare
  v_parent j_parent;
  v_child j_child;
begin
  v_child = new j_child(10, 20);
  v_parent = v_child;
  print v_parent.get_value();
end;
/
create table java_inherit_table(
  id int primary key,
  obj j_parent
);
insert into java_inherit_table values(1, j_parent(10));
insert into java_inherit_table values(2, j_child(20, 30));
select id, t.obj.base_value, t.obj.get_value() from java_inherit_table t order by id;
create or replace java abstract class j_abstract
{
  int value;
  public abstract int calculate();
}
/
create or replace java class j_concrete extends j_abstract
{
  int extra;
  public override int calculate()
  {
    return value + extra;
  }
}

  v_base j_abstract;
  v_child j_concrete;

  v_base = v_child;
  print v_base.calculate();
end;

create or replace java class j_overload
{
  int value;
  public int calculate(int a)
  {
    return a;
  }
  public int calculate(int a, int b)
  {
    return a + b;
  }
  public varchar calculate(varchar a)
  {
    return a;
  }
}
/
select j_overload().calculate(7);
select j_overload().calculate(7, 8);
select j_overload().calculate('DM8');
create or replace java class j_constructor
{
  int value;
  varchar name;
  j_constructor()
  {
    this.value = 1;
    this.name = 'DEFAULT';
  }
  j_constructor(int p_value)
  {
    this.value = p_value;
    this.name = 'ONE_ARG';
  }
  j_constructor(int p_value, varchar p_name)
  {
    this.value = p_value;
    this.name = p_name;
  }
  public varchar describe()
  {
    return name;
  }
}
/
select j_constructor().describe();
select j_constructor(23).describe();
select j_constructor(24, 'TWO_ARG').describe();
create or replace java class j_static_members
{
  private int hidden_value;
  public static int twice(int value)
  {
    return value * 2;
  }
  private int hidden()
  {
    return hidden_value;
  }
}

create or replace java class j_resource_parent
{
  public int read_value()
  {
    int v_value;
    select max(id) into v_value from java_parent_source;
    return v_value;
  }
}
/
create or replace java class j_resource_child extends j_resource_parent
{
  int child_marker;
  public override int read_value()
  {
    int v_value;
    select max(id) into v_value from java_child_source where id > 0;
    return v_value;
  }
}
/
select j_resource_child(1).read_value();
drop class j_resource_child;
drop class j_resource_parent;
