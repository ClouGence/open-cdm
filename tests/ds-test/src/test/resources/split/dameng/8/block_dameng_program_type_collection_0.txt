/

/
DECLARE V INDEX_INT; K INT; BEGIN V(1)='A'; V(3)='C'; K=V.FIRST; PRINT V(K); K=V.NEXT(K); PRINT V(K); V.DELETE(1,2); PRINT V.COUNT; END;
/

/
DECLARE SUBTYPE COUNTER IS NUMBER(5) NOT NULL; V COUNTER=99999; BEGIN PRINT V; END;
/
DECLARE C CURSOR; V_NAME TYPE_BASE.NAME%TYPE; BEGIN OPEN C FOR SELECT NAME FROM TYPE_BASE WHERE ID=1; FETCH C INTO V_NAME; PRINT V_NAME; CLOSE C; END;
/
DECLARE V INDEX_VARCHAR_OVER; BEGIN V('A')=1; PRINT V('A'); END;
