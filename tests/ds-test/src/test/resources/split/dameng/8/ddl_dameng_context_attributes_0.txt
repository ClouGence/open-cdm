  procedure set_value(namespace_name varchar, attribute_name varchar, attribute_value varchar);
  procedure set_value_as(namespace_name varchar, attribute_name varchar, attribute_value varchar,

  procedure clear_all(namespace_name varchar);
end context_attr_pkg;

create or replace package body context_attr_pkg as
  procedure set_value(namespace_name varchar, attribute_name varchar, attribute_value varchar) as
  begin
    dbms_session.set_context(namespace_name, attribute_name, attribute_value);
  end;

    username varchar, client_id varchar) as

  end;
  procedure clear_value(namespace_name varchar, client_id varchar, attribute_name varchar) as
  begin
    dbms_session.clear_context(namespace_name, client_id, attribute_name);
  end;
  procedure clear_all(namespace_name varchar) as
  begin
    dbms_session.clear_all_context(namespace_name);
  end;
end context_attr_pkg;
