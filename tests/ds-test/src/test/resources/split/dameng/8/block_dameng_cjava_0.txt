  }
}

create or replace java class j_math
{
  int a;
  public static int add_two(int a, int b)
  {
    return a + b;
  }
  public int add_three(int a, int b, int c)
  {
    return a + b + c;
  }
}
/
select j_math.add_two(1, 2);
select j_math().add_three(1, 2, 3);
create table cjava_class_table(
  id int primary key,
  obj j_math
);
insert into cjava_class_table values(j_math.add_two(1, 2), j_math(1));
insert into cjava_class_table values(j_math().add_three(1, 2, 3), j_math(2));
select id, t.obj.a from cjava_class_table t order by id;

create or replace java class j_cursor_reader
{
  varchar v_name;
  int v_id;
  int first_id()
  {
    cursor csr for select id, name from cjava_source order by id;
    open cursor csr;
    fetch cursor csr into v_id, v_name;
    close cursor csr;
    return v_id;
  }
}
