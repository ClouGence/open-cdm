-- POSITIONAL_STATE
DECLARE
  V_CLIENT VARCHAR2(128); V_MODULE VARCHAR2(128); V_ACTION VARCHAR2(128);
BEGIN
  DBMS_APPLICATION_INFO.SET_CLIENT_INFO('CODEX_CLIENT');
  DBMS_APPLICATION_INFO.SET_MODULE('CODEX_MODULE','ACTION_1');
  DBMS_APPLICATION_INFO.SET_ACTION('ACTION_2');
  DBMS_APPLICATION_INFO.READ_CLIENT_INFO(V_CLIENT);
  DBMS_APPLICATION_INFO.READ_MODULE(V_MODULE,V_ACTION);
  IF V_CLIENT <> 'CODEX_CLIENT' OR V_MODULE <> 'CODEX_MODULE' OR V_ACTION <> 'ACTION_2' THEN
    RAISE_APPLICATION_ERROR(-20001,'POSITIONAL_STATE');
  END IF;
END;
/
-- NULL_RESET
DECLARE
  V_CLIENT VARCHAR2(128); V_MODULE VARCHAR2(128); V_ACTION VARCHAR2(128);
BEGIN
  DBMS_APPLICATION_INFO.SET_CLIENT_INFO('VALUE');
  DBMS_APPLICATION_INFO.SET_MODULE('MODULE','ACTION');
  DBMS_APPLICATION_INFO.SET_CLIENT_INFO(NULL);
  DBMS_APPLICATION_INFO.SET_MODULE(NULL,NULL);
  DBMS_APPLICATION_INFO.SET_ACTION(NULL);
  DBMS_APPLICATION_INFO.READ_CLIENT_INFO(V_CLIENT);
  DBMS_APPLICATION_INFO.READ_MODULE(V_MODULE,V_ACTION);
  IF NVL(LENGTHB(V_CLIENT),0) <> 0 OR NVL(LENGTHB(V_MODULE),0) <> 0
    OR NVL(LENGTHB(V_ACTION),0) <> 0 THEN
    RAISE_APPLICATION_ERROR(-20001,'NULL_RESET');
  END IF;
END;
/
