create partition group pg_hh partition by hash(int) subpartition by hash(int) subpartition template subpartitions 2 partitions 2 storage(on ts_1,ts_2);
create partition group pg_hl partition by hash(int) subpartition by list(int) subpartition template(subpartition sa values(1),subpartition sd values(default)) partitions 2 storage(on ts_1,ts_2);
create partition group pg_lr partition by list(int) subpartition by range(int) subpartition template(subpartition s1 values less than(10),subpartition smax values less than(maxvalue))(partition p1 values(1),partition pd values(default)) storage(on ts_1,ts_2);
create partition group pg_lh partition by list(int) subpartition by hash(int) subpartition template subpartitions 2(partition p1 values(1),partition pd values(default)) storage(on ts_1,ts_2);
create partition group pg_ll partition by list(int) subpartition by list(int) subpartition template(subpartition sa values(1),subpartition sd values(default))(partition p1 values(1),partition pd values(default)) storage(on ts_1,ts_2);
