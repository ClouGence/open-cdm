SELECT JSON_TYPE(DOC) AS JSON_KIND FROM JSON_STRICT ORDER BY ID;
SELECT DOC::JSON->'a' AS A_VALUE FROM JSON_LAX;
SELECT JSON_QUERY(DOC, '$' WITH WRAPPER) AS WRAPPED_DOC FROM JSON_UNIQUE;
SELECT ID FROM JSON_STRICT WHERE DOC IS JSON(STRICT) ORDER BY ID;
SELECT ID FROM JSON_LAX WHERE DOC IS JSON(LAX);
SELECT ID FROM JSON_NONUNIQUE WHERE DOC IS JSON WITHOUT UNIQUE KEYS;
SELECT ID FROM JSON_UNIQUE WHERE DOC IS JSON WITH UNIQUE KEYS;
SELECT ID FROM JSON_LONG WHERE DOC IS NOT JSON;
SELECT TABLE_NAME, COLUMN_NAME, FORMAT, DATA_TYPE FROM USER_JSON_COLUMNS ORDER BY TABLE_NAME;
SELECT JSON_EXTRACT('{"a":[1,2]}', '$.a[1]');
SELECT JSON_UNQUOTE('"a\\tb"');
SELECT * FROM JSON_OBJECT_KEYS('{"b":2,"a":1}');
SELECT * FROM JSONB_EACH('{"a":1,"b":"x\\ty"}');
SELECT * FROM JSONB_EACH_TEXT('{"a":1,"b":"x\\ty"}');
SELECT * FROM JSONB_ARRAY_ELEMENTS('[1,true,null,"x\\ty"]');
SELECT * FROM JSONB_ARRAY_ELEMENTS_TEXT('[1,true,null,"x\\ty"]');
SELECT JSONB_STRIP_NULLS('[null,{"a":null,"b":1}]');
SELECT JSONB_STRIP_NULLS(NULL);
SELECT JSONB_BUILD_OBJECT('a',1,'b',true,'c',NULL);
SELECT JSONB_BUILD_ARRAY(1,'a',true,NULL,'{"x":1}'::JSONB);
SELECT JSONB_AGG(ID) FROM (SELECT ID FROM JSON_STRICT ORDER BY ID);
SELECT JSONB_OBJECT_AGG(CAST(ID AS VARCHAR), DOC) FROM JSON_STRICT;
SELECT JSONB_TYPEOF('"text"');
SELECT JSONB_TYPEOF('1');
SELECT JSONB_TYPEOF('true');
SELECT JSONB_TYPEOF('null');
SELECT JSONB_TYPEOF('{}');
SELECT JSONB_TYPEOF('[]');
SELECT '[1,true,"abc"]'::JSON->0;
SELECT '{"a":"bcd","e":[1,2,3]}'::JSON->'e';
SELECT '[1,true,"abc"]'::JSONB->>2;
SELECT '{"a":"bcd","e":[1,2,3]}'::JSONB->>'a';
SELECT '{"a":1,"b":2}'::JSONB-'a';
SELECT '[1,{"a":1},"c"]'::JSONB-1;
SELECT '[1,{"a":1},"c"]'::JSONB-'a';
SELECT '{"a":1}'::JSONB-NULL;
SELECT '{"a":1,"b":2}'::JSONB @> '{"b":2}'::JSONB;
SELECT '{"a":1,"b":2}'::JSONB @> '{"b":2,"b":2}';
