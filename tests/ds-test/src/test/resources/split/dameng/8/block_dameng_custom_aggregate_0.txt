create type dm8_max_impl as object(
  max_value number,
  static function ODCIAggregateInitialize(sctx in out dm8_max_impl) return number,
  member function ODCIAggregateIterate(self in out dm8_max_impl, value in number) return number,
  member function ODCIAggregateTerminate(self in dm8_max_impl,
    return_value out number, flags in number) return number
);
/
create type body dm8_max_impl is
  static function ODCIAggregateInitialize(sctx in out dm8_max_impl) return number is
  begin
    sctx := dm8_max_impl(-999999999);
    return ODCIConst.Success;
  end;
  member function ODCIAggregateIterate(self in out dm8_max_impl,
    value in number) return number is
  begin
    if value > self.max_value then
      self.max_value := value;
    end if;
    return ODCIConst.Success;
  end;
  member function ODCIAggregateTerminate(self in dm8_max_impl,
    return_value out number, flags in number) return number is
  begin
    return_value := self.max_value;
    return ODCIConst.Success;
  end;
end;
/
