SELECT TINY_MIN, TINY_MAX, SMALL_MIN, SMALL_MAX, INT_MIN, INT_MAX, BIG_MIN, BIG_MAX
FROM CODEX_DM8_TYPE_INTEGER WHERE ID = 1;
SELECT MIN(TINY_MIN), MAX(BIG_MAX), SUM(SMALL_MAX) FROM CODEX_DM8_TYPE_INTEGER;
SELECT N_MIN, N_MAX, N_SCALE_MIN, N_SCALE_MAX, N_ROUND,
       F_MIN, F_REAL, F_DOUBLE, F_MAX, D_MAX, DP_MAX, R
FROM CODEX_DM8_TYPE_NUMERIC WHERE ID = 1;
SELECT LENGTH(C_MIN), LENGTH(C_ALIAS), LENGTH(V_MIN), LENGTH(V_CHAR), LENGTH(V_ALIAS)
FROM CODEX_DM8_TYPE_CHAR WHERE ID = 1;
SELECT LENGTH(CAST(REPEAT('C', 32767) AS CHAR(32767))),
       LENGTH(CAST(REPEAT('V', 32767) AS VARCHAR(32767)))
FROM DUAL;
SELECT LENGTHB(B_MIN), LENGTHB(VB_MIN), LENGTHB(RAW_MAX)
FROM CODEX_DM8_TYPE_BINARY WHERE ID = 1;
SELECT DATE_MIN, DATE_MAX, TIME_MIN, TIME_MAX, TS_MIN, TS_MAX
FROM CODEX_DM8_TYPE_DATETIME WHERE ID = 1;
SELECT DATE '1582-10-04' + 1 AS CALENDAR_NEXT_DAY FROM DUAL;

FROM CODEX_DM8_TYPE_INTERVAL WHERE ID = 1;
SELECT T_NEG, T_POS, TS_NEG, TS_POS, TS_LOCAL FROM CODEX_DM8_TYPE_TZ WHERE ID = 1;
SELECT CAST(TS_NEG AS TIMESTAMP(9)), CAST(TS_LOCAL AS TIMESTAMP(9) WITH TIME ZONE)
FROM CODEX_DM8_TYPE_TZ WHERE ID = 1;
