select o.id, c.name
from app.orders o
join app.customer c on c.id = o.customer_id
where o.order_total > 10;

with recent_orders as (
  select id, customer_id from app.orders
)
select r.id, c.name
from recent_orders r
join app.customer c on c.id = r.customer_id;

create table app.order_snapshot as
select o.id, c.name from app.orders o join app.customer c on c.id = o.customer_id;

create table app.customer (
  id int identity primary key,
  name varchar(100) not null,
  status char(1) default 'A',
  constraint uk_customer_name unique(name)
);

create index idx_orders_customer on app.orders(customer_id);

create view app.active_customer_view as select id, name from app.customer where status = 'A';

create sequence app.seq_orders start with 1 increment by 1 cache 20;

create or replace procedure app.proc_refresh_customer(p_id int) as
begin
  update app.customer set status = 'A' where id = p_id;
end;
/

create or replace function app.fn_customer_name(p_id int) return varchar as
  v_name varchar(100);
begin
  select name into v_name from app.customer where id = p_id;
  return v_name;
end;
/

create or replace trigger app.trg_customer_bi
before insert on app.customer
for each row
begin
  if :new.status is null then
    :new.status := 'A';
  end if;
end;
/

insert into app.customer(id, name, status) values(1, 'Alice', 'A');
update app.customer set status = 'I' where id = 1;
delete from app.customer where id = 1;
comment on column app.customer.name is 'customer name';
alter table app.customer add email varchar(200);
create tablespace ts_dm_test datafile 'ts_dm_test.dbf' size 128;
drop table app.customer;
