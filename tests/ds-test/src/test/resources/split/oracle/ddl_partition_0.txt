CREATE TABLE sales_range (
    sale_id NUMBER,
    amount NUMBER
)
PARTITION BY RANGE (sale_id) (
    PARTITION p100 VALUES LESS THAN (100),
    PARTITION pmax VALUES LESS THAN (MAXVALUE)
);

CREATE TABLE sales_list (
    sale_id NUMBER,
    region VARCHAR2(20)
)
PARTITION BY LIST (region) (
    PARTITION p_east VALUES ('EAST'),
    PARTITION p_west VALUES ('WEST')
);

CREATE TABLE sales_hash (
    sale_id NUMBER,
    amount NUMBER
)
PARTITION BY HASH (sale_id)
PARTITIONS 4;

CREATE TABLE sales_range_sub (
    sale_id NUMBER,
    region VARCHAR2(20),
    amount NUMBER
)
PARTITION BY RANGE (sale_id)
SUBPARTITION BY LIST (region) (
    PARTITION p100 VALUES LESS THAN (100) (
        SUBPARTITION p100_east VALUES ('EAST'),
        SUBPARTITION p100_west VALUES ('WEST')
    ),
    PARTITION pmax VALUES LESS THAN (MAXVALUE) (
        SUBPARTITION pmax_east VALUES ('EAST'),
        SUBPARTITION pmax_west VALUES ('WEST')
    )
);
