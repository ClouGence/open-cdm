EXPLAIN TABLE utility_audit.maint_a;
SELECT t1.*,dt.c FROM jt AS t1 CROSS JOIN LATERAL (SELECT COUNT(*) AS c FROM jt AS t2 LEFT JOIN jt AS t3 ON t3.a>t2.a-t1.a) AS dt;
SELECT id, grp, v FROM t1 NATURAL INNER JOIN t2;
SELECT t1.id FROM t1 JOIN t2 JOIN t3 ON t2.id = t3.id ON t1.id = t2.id;
SELECT t.id, d.n FROM t JOIN LATERAL (SELECT t.id + 1 AS n) AS d ON TRUE ORDER BY t.id;
SELECT d2.v FROM (SELECT * FROM split_scalar80_d.q) AS d1 JOIN (SELECT * FROM split_scalar80_d.q) AS d2 JOIN split_scalar80_d.q AS q1 ON d2.v<(SELECT MAX(q2.v) FROM split_scalar80_d.q AS q2) ON TRUE;
(TABLE t ORDER BY id LIMIT 2) INTERSECT (TABLE t2 ORDER BY id LIMIT 2);
TABLE src ORDER BY id LIMIT 1;
SELECT * FROM split_sample.t TABLESAMPLE BERNOULLI (0.5);
SELECT s.id FROM split_sample.t PARTITION (p0) AS s USE INDEX (idx_grp) TABLESAMPLE BERNOULLI (50) LEFT JOIN split_sample.t2 TABLESAMPLE SYSTEM (100) ON s.id = t2.id;
SELECT * FROM split_sample.t TABLESAMPLE SYSTEM (10);
SELECT o.id, d.id, d.rn
  FROM t1 AS o
  JOIN LATERAL (
      SELECT i.id, ROW_NUMBER() OVER (ORDER BY i.v DESC) AS rn
        FROM t2 AS i
       WHERE i.grp = o.grp
       ORDER BY i.v DESC
       LIMIT 2
  ) AS d ON TRUE
 WHERE d.rn = 1
 ORDER BY o.id
 LIMIT 5;
SELECT o.id, d.total
  FROM t1 AS o
  JOIN LATERAL (
      SELECT SUM(i.v) AS total, GROUPING(i.grp) AS g
        FROM t2 AS i
       WHERE i.grp = o.grp
       GROUP BY GROUPING SETS ((i.grp), ())
      HAVING SUM(i.v) > 0
       ORDER BY g
       LIMIT 1
  ) AS d ON TRUE
 ORDER BY o.id
 LIMIT 5;
