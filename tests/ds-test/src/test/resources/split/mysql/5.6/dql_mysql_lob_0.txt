SELECT HEX(b),OCTET_LENGTH(mb),CHAR_LENGTH(t),LENGTH(lt)
FROM lob_family
WHERE HEX(b) LIKE '41%' AND t LIKE 'text%'
ORDER BY t,b
LIMIT 5;
SELECT t,COUNT(*),MIN(HEX(b)),MAX(CHAR_LENGTH(lt))
FROM lob_family
GROUP BY t
HAVING COUNT(*) >= 1
ORDER BY t;
SELECT outer_lob.t
FROM lob_family AS outer_lob
WHERE EXISTS (
  SELECT 1 FROM lob_family AS inner_lob
  WHERE inner_lob.b=outer_lob.b AND inner_lob.t=outer_lob.t
);
SELECT a.t,HEX(b.b)
FROM lob_family AS a
JOIN lob_family AS b ON a.b=b.b AND a.t=b.t
WHERE a.mb IS NOT NULL;
SELECT CASE WHEN b IS NULL THEN CAST('none' AS CHAR) ELSE HEX(b) END,
       COALESCE(t,mt,lt),
       NULLIF(b,X'00')
FROM lob_family;
SELECT t FROM lob_family
UNION ALL
SELECT CAST(b AS CHAR CHARACTER SET utf8mb4) FROM lob_family;
SELECT (SELECT (b <> 1) FROM lob_family LIMIT 1) <> ALL
       (SELECT 1681007452 FROM lob_family)
FROM lob_family;
