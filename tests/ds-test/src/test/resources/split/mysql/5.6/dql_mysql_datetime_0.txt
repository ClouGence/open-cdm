SELECT d.id,d.dt6,a.modified_at FROM datetime_widths AS d JOIN temporal_auto AS a ON d.dt6=a.created_at OR d.ts6=a.modified_at;
SELECT id,dt6,ts6 FROM datetime_widths WHERE(dt6,ts6) IN(SELECT created_at,modified_at FROM temporal_auto WHERE explicit_at IS NOT NULL);
SELECT CASE WHEN id=1 THEN dt1 ELSE ts1 END AS temporal_case,IF(id>0,dt3,ts3) AS temporal_if,COALESCE(dt6,ts6,CURRENT_TIMESTAMP(6)) AS temporal_coalesce,NULLIF(dt5,ts5) AS temporal_nullif,LEAST(dt4,ts4) AS temporal_least,GREATEST(dt2,ts2) AS temporal_greatest FROM datetime_widths;
SELECT dt6 AS temporal_value FROM datetime_widths UNION ALL SELECT modified_at FROM temporal_auto;
