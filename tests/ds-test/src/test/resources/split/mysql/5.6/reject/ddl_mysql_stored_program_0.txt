CREATE PROCEDURE proc_stacked_diag()
BEGIN
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    GET STACKED DIAGNOSTICS CONDITION 1 @stack_state = RETURNED_SQLSTATE, @stack_message = MESSAGE_TEXT;
    RESIGNAL;
  END;
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'stacked diagnostics';
END;
CREATE EVENT split56.ev_replica ON SCHEDULE EVERY 1 DAY DISABLE ON REPLICA DO SET @event_seen = 1;
CREATE FUNCTION IF NOT EXISTS f_ifne() RETURNS INT DETERMINISTIC NO SQL RETURN 0;
CREATE PROCEDURE IF NOT EXISTS p_ifne() BEGIN END;
CREATE DEFINER = CURRENT_USER TRIGGER IF NOT EXISTS split_trigger.trg_bi
BEFORE INSERT ON split_trigger.src
FOR EACH ROW
SET NEW.updated_at = CURRENT_TIMESTAMP;
CREATE TRIGGER split56.trg_order_follows BEFORE INSERT ON split56.trigger_src FOR EACH ROW FOLLOWS trg_order_base SET NEW.note = 'follows';
CREATE PROCEDURE p11() BEGIN DECLARE EXIT HANDLER FOR SQLEXCEPTION BEGIN GET CURRENT DIAGNOSTICS CONDITION 1 @m1=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m2=MESSAGE_TEXT; SELECT 1; GET CURRENT DIAGNOSTICS @n=NUMBER; GET STACKED DIAGNOSTICS CONDITION 1 @m3=MESSAGE_TEXT; END; DROP TABLE missing_t; END;
CREATE PROCEDURE p12() BEGIN DECLARE CONTINUE HANDLER FOR SQLWARNING BEGIN GET CURRENT DIAGNOSTICS CONDITION 1 @m1=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m2=MESSAGE_TEXT; RESIGNAL SET MYSQL_ERRNO=9999,MESSAGE_TEXT='mapped'; GET CURRENT DIAGNOSTICS CONDITION 1 @m3=MESSAGE_TEXT; GET STACKED DIAGNOSTICS CONDITION 1 @m4=MESSAGE_TEXT; END; SIGNAL SQLSTATE '01001'; END;
CREATE PROCEDURE p15() BEGIN DECLARE done BOOL DEFAULT FALSE; DECLARE n INT; DECLARE c CURSOR FOR WITH RECURSIVE seq(n) AS (SELECT 1 UNION ALL SELECT n+1 FROM seq WHERE n<3) SELECT n FROM seq; DECLARE CONTINUE HANDLER FOR NOT FOUND SET done=TRUE; OPEN c; FETCH c INTO n; CLOSE c; END;
CREATE PROCEDURE p_jdv() BEGIN CREATE JSON DUALITY VIEW dv_in_proc AS SELECT JSON_DUALITY_OBJECT('_id':c1) FROM t1; DROP VIEW dv_in_proc; END;
