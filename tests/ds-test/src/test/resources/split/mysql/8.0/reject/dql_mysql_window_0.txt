WITH totals AS (
    SELECT grp, SUM(v) AS total FROM t2 GROUP BY grp
)
SELECT t.id, totals.total,
       ROW_NUMBER() OVER (PARTITION BY t.grp ORDER BY t.v DESC) AS rn
  FROM t1 AS t
  JOIN totals ON totals.grp = t.grp
 QUALIFY rn = 1 AND totals.total > 0
 ORDER BY t.id
 LIMIT 5;
SELECT o.id, d.id
  FROM t1 AS o
  JOIN LATERAL (
      SELECT i.id, ROW_NUMBER() OVER (ORDER BY i.v DESC) AS rn
        FROM t2 AS i
       WHERE i.grp = o.grp
       QUALIFY rn = 1
  ) AS d ON TRUE
 ORDER BY o.id
 LIMIT 5;
WITH sampled AS (
    SELECT id, grp, v FROM t1 TABLESAMPLE SYSTEM (25)
)
SELECT id, ROW_NUMBER() OVER (PARTITION BY grp ORDER BY v DESC) AS rn
  FROM sampled
 QUALIFY rn <= 2
 ORDER BY id
 LIMIT 5;
