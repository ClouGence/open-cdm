CREATE PROCEDURE proc_compound_common(IN p_limit INT, OUT p_total INT)
COMMENT 'compound coverage'
LANGUAGE SQL
MODIFIES SQL DATA
SQL SECURITY INVOKER
BEGIN
  DECLARE v_id INT DEFAULT 0;
  DECLARE v_done BOOL DEFAULT FALSE;
  DECLARE cur_ids CURSOR FOR SELECT id FROM proc_src ORDER BY id;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_done = TRUE;
  SET p_total = 0;
  OPEN cur_ids;
  read_loop: LOOP
    FETCH NEXT FROM cur_ids INTO v_id;
    IF v_done THEN
      LEAVE read_loop;
    END IF;
    SET p_total = p_total + 1;
    CASE
      WHEN p_total < p_limit THEN SET v_id = v_id + 0;
      ELSE SET v_id = v_id;
    END CASE;
    IF p_total < p_limit THEN
      ITERATE read_loop;
    END IF;
  END LOOP read_loop;
  CLOSE cur_ids;
  WHILE p_total < p_limit DO
    SET p_total = p_total + 1;
  END WHILE;
  REPEAT
    SET p_total = p_total - 1;
  UNTIL p_total <= p_limit
  END REPEAT;
END;
CREATE FUNCTION fn_compound_common(p_value INT)
RETURNS INT
COMMENT 'function coverage'
LANGUAGE SQL
DETERMINISTIC
NO SQL
SQL SECURITY INVOKER
BEGIN
  DECLARE v_result INT DEFAULT 0;
  IF p_value IS NULL THEN
    SET v_result = 0;
  ELSE
    SET v_result = p_value + 1;
  END IF;
  RETURN v_result;
END;

CREATE PROCEDURE load_locale_format_table()
BEGIN
  DECLARE locale_list VARCHAR(1000) DEFAULT '
  es_AR,es_BO,es_CL,es_CO,es_CR,es_DO,es_EC,es_ES,es_GT,es_HN,
  es_MX,es_NI,es_PA,es_PE,es_PR,es_PY,es_SV,es_US,es_UY,es_VE';
  SET @fmt_stmt = 'INSERT INTO locale_format VALUES
                   (?, FORMAT(12131254123412541,2,?));';
  PREPARE stmt FROM @fmt_stmt;
  WHILE locale_list != '' DO
    SET @locale =
          TRIM(REPLACE((SUBSTRING_INDEX(locale_list, ',', 1)), '\n',''));
    EXECUTE stmt USING @locale, @locale;
    IF LOCATE(',', locale_list) > 0 THEN
      SET locale_list =
      SUBSTRING(locale_list, LOCATE(',', locale_list) + 1);
    ELSE
      SET locale_list = '';
    END IF;
  END WHILE;
  DEALLOCATE PREPARE stmt;
END;

[CREATE_PROG_OBJ|COMMENT_PROG_OBJ(BLOCK,PROGRAM_CONTROL,SELECT)] CREATE PROCEDURE proc_compound_common(IN p_limit INT, OUT p_total INT)
COMMENT 'compound coverage'
LANGUAGE SQL
MODIFIES SQL DATA
SQL SECURITY INVOKER
BEGIN
  DECLARE v_id INT DEFAULT 0;
  DECLARE v_done BOOL DEFAULT FALSE;
  DECLARE cur_ids CURSOR FOR SELECT id FROM proc_src ORDER BY id;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_done = TRUE;
  SET p_total = 0;
  OPEN cur_ids;
  read_loop: LOOP
    FETCH NEXT FROM cur_ids INTO v_id;
    IF v_done THEN
      LEAVE read_loop;
    END IF;
    SET p_total = p_total + 1;
    CASE
      WHEN p_total < p_limit THEN SET v_id = v_id + 0;
      ELSE SET v_id = v_id;
    END CASE;
    IF p_total < p_limit THEN
      ITERATE read_loop;
    END IF;
  END LOOP read_loop;
  CLOSE cur_ids;
  WHILE p_total < p_limit DO
    SET p_total = p_total + 1;
  END WHILE;
  REPEAT
    SET p_total = p_total - 1;
  UNTIL p_total <= p_limit
  END REPEAT;
END;
----------
[CREATE_PROG_OBJ|COMMENT_PROG_OBJ(BLOCK,PROGRAM_CONTROL)] CREATE FUNCTION fn_compound_common(p_value INT)
RETURNS INT
COMMENT 'function coverage'
LANGUAGE SQL
DETERMINISTIC
NO SQL
SQL SECURITY INVOKER
BEGIN
  DECLARE v_result INT DEFAULT 0;
  IF p_value IS NULL THEN
    SET v_result = 0;
  ELSE
    SET v_result = p_value + 1;
  END IF;
  RETURN v_result;
END;

[CREATE_EVENT|COMMENT_EVENT(BLOCK,PROGRAM_CONTROL,SELECT,UPDATE,MERGE,SESSION_VARIABLE_RW)] CREATE EVENT split_event.ev_compound
ON SCHEDULE EVERY (1 + 1) DAY
STARTS CURRENT_TIMESTAMP + INTERVAL 1 HOUR
ENDS CURRENT_TIMESTAMP + INTERVAL 7 DAY
ON COMPLETION PRESERVE
DISABLE
COMMENT 'compound event'
DO event_block: BEGIN
  DECLARE done BOOLEAN DEFAULT FALSE;
  DECLARE v_id INT;
  DECLARE cur CURSOR FOR SELECT id FROM split_event.event_log;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
  OPEN cur;
  read_loop: LOOP
    FETCH cur INTO v_id;
    IF done THEN
      LEAVE read_loop;
    END IF;
    UPDATE split_event.event_log SET note = CONCAT('event-', v_id) WHERE id = v_id;
  END LOOP read_loop;
  CLOSE cur;
  CASE
    WHEN EXISTS (SELECT 1 FROM split_event.event_log WHERE id = 1) THEN
      INSERT INTO split_event.event_log VALUES (2, 'created') ON DUPLICATE KEY UPDATE note = VALUES(note);
    ELSE
      SET @event_case = 0;
  END CASE;
END event_block;

[CREATE_PROG_OBJ(BLOCK,PROGRAM_CONTROL,SESSION_VARIABLE_RW,UNSAFE)] CREATE PROCEDURE load_locale_format_table()
BEGIN
  DECLARE locale_list VARCHAR(1000) DEFAULT '
  es_AR,es_BO,es_CL,es_CO,es_CR,es_DO,es_EC,es_ES,es_GT,es_HN,
  es_MX,es_NI,es_PA,es_PE,es_PR,es_PY,es_SV,es_US,es_UY,es_VE';
  SET @fmt_stmt = 'INSERT INTO locale_format VALUES
                   (?, FORMAT(12131254123412541,2,?));';
  PREPARE stmt FROM @fmt_stmt;
  WHILE locale_list != '' DO
    SET @locale =
          TRIM(REPLACE((SUBSTRING_INDEX(locale_list, ',', 1)), '\n',''));
    EXECUTE stmt USING @locale, @locale;
    IF LOCATE(',', locale_list) > 0 THEN
      SET locale_list =
      SUBSTRING(locale_list, LOCATE(',', locale_list) + 1);
    ELSE
      SET locale_list = '';
    END IF;
  END WHILE;
  DEALLOCATE PREPARE stmt;
END;
