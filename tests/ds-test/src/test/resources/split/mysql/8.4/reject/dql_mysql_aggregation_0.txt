SELECT i, SUM(j) AS total, GROUPING(i) AS gi, RANK() OVER (ORDER BY SUM(j) DESC) AS rnk FROM (SELECT 1 AS i, 2 AS j UNION ALL SELECT 1, 3) AS t GROUP BY GROUPING SETS ((i), ()) HAVING SUM(j) > 0 ORDER BY i;

WITH RECURSIVE seq(n) AS (
    SELECT 1
    UNION ALL
    SELECT n + 1 FROM seq WHERE n < 4
)
SELECT n % 2 AS grp, GROUPING(n % 2) AS g, SUM(n) AS total,
       RANK() OVER (ORDER BY SUM(n) DESC) AS rnk
  FROM seq
 GROUP BY GROUPING SETS ((n % 2), ())
HAVING SUM(n) > 0
 ORDER BY g, grp
 LIMIT 3;
SELECT grp, SUM(v) AS total, GROUPING(grp) AS g,
       RANK() OVER (ORDER BY SUM(v) DESC) AS rnk
  FROM t1
 GROUP BY GROUPING SETS ((grp), ())
HAVING SUM(v) > 0
 QUALIFY rnk <= 2
 ORDER BY g, grp
 LIMIT 3;
