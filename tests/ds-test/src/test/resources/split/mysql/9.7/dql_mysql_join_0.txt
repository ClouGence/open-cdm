SELECT DISTINCT
       projects.project_id,
       projects.project_name,
       projects.client_ptr,
       projects.comments,
       SUM(budgets.amount_received) + SUM(budgets.adjustment) AS total_budget
FROM join_crash_periods AS client_period,
     join_crash_periods AS project_period,
     join_crash_budgets AS budgets
LEFT JOIN join_crash_projects AS projects
  ON budgets.project_ptr = projects.project_id
 AND budgets.date_received <= '2001-03-22 14:15:09'
LEFT JOIN join_crash_clients AS clients
  ON clients.client_id = projects.client_ptr
WHERE client_period.period_type = 'client_table'
  AND client_period.period_key = clients.client_id
  AND (client_period.start_date <= '2001-03-22 14:15:09' OR ISNULL(client_period.start_date))
  AND (client_period.end_date > '2001-03-21 14:15:09' OR ISNULL(client_period.end_date))

  AND project_period.period_key = projects.project_id

  AND (project_period.end_date > '2001-03-21 14:15:09' OR ISNULL(project_period.end_date))
GROUP BY clients.client_id,
         projects.project_id,
         client_period.period_id,
         project_period.period_id

         projects.project_name ASC;
