WITH RECURSIVE tree (id,parent_id,depth,path) AS (
  SELECT id,parent_id,0,CAST(id AS CHAR(100)) FROM split97_recursive.nodes WHERE parent_id IS NULL
  UNION ALL
  SELECT c.id,c.parent_id,p.depth+1,CONCAT(p.path,'/',c.id)
  FROM split97_recursive.nodes AS c JOIN tree AS p ON c.parent_id=p.id
  WHERE p.depth<5
), scored AS (
  SELECT id,parent_id,depth,path,
         SUM(depth) OVER (PARTITION BY parent_id ORDER BY id ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_depth,
         (SELECT COUNT(*) FROM split97_recursive.nodes AS x
          WHERE x.parent_id=tree.id
            AND EXISTS (SELECT 1 FROM split97_recursive.nodes AS y WHERE y.id=x.id AND y.score>=x.score)) AS child_count
  FROM tree
)
SELECT COALESCE(parent_id,-1) AS parent_key,
       MAX(running_depth) AS max_running,
       SUM(child_count) AS total_children,
       CASE WHEN EXISTS (
              SELECT 1 FROM scored AS s2
              WHERE s2.parent_id <=> scored.parent_id AND s2.child_count>0
            ) THEN 'branch' ELSE 'leaf' END AS kind
FROM scored
WHERE id IN (
  SELECT n.id FROM split97_recursive.nodes AS n
  WHERE n.id=scored.id OR n.parent_id=scored.id
)
GROUP BY parent_id
HAVING SUM(child_count) >= (SELECT MIN(s3.child_count) FROM scored AS s3)
UNION ALL
SELECT -2,0,0,'sentinel'
ORDER BY parent_key
LIMIT 10;

[SELECT] WITH RECURSIVE tree (id,parent_id,depth,path) AS (
  SELECT id,parent_id,0,CAST(id AS CHAR(100)) FROM split97_recursive.nodes WHERE parent_id IS NULL
  UNION ALL
  SELECT c.id,c.parent_id,p.depth+1,CONCAT(p.path,'/',c.id)
  FROM split97_recursive.nodes AS c JOIN tree AS p ON c.parent_id=p.id
  WHERE p.depth<5
), scored AS (
  SELECT id,parent_id,depth,path,
         SUM(depth) OVER (PARTITION BY parent_id ORDER BY id ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_depth,
         (SELECT COUNT(*) FROM split97_recursive.nodes AS x
          WHERE x.parent_id=tree.id
            AND EXISTS (SELECT 1 FROM split97_recursive.nodes AS y WHERE y.id=x.id AND y.score>=x.score)) AS child_count
  FROM tree
)
SELECT COALESCE(parent_id,-1) AS parent_key,
       MAX(running_depth) AS max_running,
       SUM(child_count) AS total_children,
       CASE WHEN EXISTS (
              SELECT 1 FROM scored AS s2
              WHERE s2.parent_id <=> scored.parent_id AND s2.child_count>0
            ) THEN 'branch' ELSE 'leaf' END AS kind
FROM scored
WHERE id IN (
  SELECT n.id FROM split97_recursive.nodes AS n
  WHERE n.id=scored.id OR n.parent_id=scored.id
)
GROUP BY parent_id
HAVING SUM(child_count) >= (SELECT MIN(s3.child_count) FROM scored AS s3)
UNION ALL
SELECT -2,0,0,'sentinel'
ORDER BY parent_key
LIMIT 10;
