SELECT id,e_basic,e_basic+0,CAST(e_basic AS UNSIGNED),
               e_basic='alpha',e_basic=2
        FROM split_type_enum_set.es_core
        ORDER BY e_basic,id;
SELECT id,e_basic
        FROM split_type_enum_set.es_core
        ORDER BY CAST(e_basic AS CHAR),CONCAT(e_basic),id;
SELECT id,e_basic,s_basic
        FROM split_type_enum_set.es_core
        WHERE (
                e_basic IN ('alpha','two words')
                OR e_basic BETWEEN 'alpha' AND 'trailing '
                OR e_basic <=> NULL
              )
          AND (
                s_basic COLLATE utf8mb4_bin = 'red,blue'
                OR s_basic IN ('green','blue,red')
                OR s_basic <=> NULL
              );
SELECT id,s_basic,s_basic+0,(s_basic+0 & 1),(s_basic+0 & 4),
               FIND_IN_SET('green',s_basic),s_basic LIKE '%blue%',
               s_basic='red,blue'
        FROM split_type_enum_set.es_core
        ORDER BY s_basic+0,id;
SELECT e_basic,COUNT(*),BIT_OR(s_basic+0)
        FROM split_type_enum_set.es_core
        GROUP BY e_basic
        HAVING e_basic+0 >= 1
        ORDER BY e_basic;
SELECT a.id,a.e_basic,b.s_basic
        FROM split_type_enum_set.es_core AS a
        JOIN split_type_enum_set.es_core AS b
          ON a.e_basic=b.e_basic AND (a.s_basic+0 & b.s_basic+0) <> 0
        ORDER BY a.id,b.id;
SELECT a.id,a.e_basic
        FROM split_type_enum_set.es_core AS a
        WHERE EXISTS (
          SELECT 1 FROM split_type_enum_set.es_core AS b
          WHERE b.e_basic=a.e_basic AND FIND_IN_SET('red',b.s_basic)
        )
        ORDER BY a.id;
SELECT d.enum_index,d.set_mask,COUNT(*)
        FROM (
          SELECT e_basic+0 AS enum_index,s_basic+0 AS set_mask
          FROM split_type_enum_set.es_core
        ) AS d
        GROUP BY d.enum_index,d.set_mask
        HAVING BIT_OR(d.set_mask) > 0;
SELECT e_basic AS v FROM split_type_enum_set.es_core
        UNION ALL
        SELECT s_basic FROM split_type_enum_set.es_core
        ORDER BY v;
SELECT id,
               CASE e_basic WHEN 'alpha' THEN e_case ELSE e_basic END AS enum_case,
               CASE WHEN FIND_IN_SET('red',s_basic) THEN s_case ELSE s_basic END AS set_case
        FROM split_type_enum_set.es_core;
SELECT
          (SELECT MAX(e_basic+0) FROM split_type_enum_set.es_core) AS max_enum_index,
          (SELECT BIT_OR(s_basic+0) FROM split_type_enum_set.es_core) AS combined_set_mask;
WITH enum_set_cte AS (
          SELECT id,e_basic,e_basic+0 AS e_index,s_basic,s_basic+0 AS s_mask
          FROM split_type_enum_set.es_core
        ),
        filtered AS (
          SELECT * FROM enum_set_cte WHERE (s_mask & 1) = 1
        )
        SELECT id,e_basic,s_basic
        FROM filtered
        WHERE e_index >= 1
        ORDER BY e_index,id;
WITH RECURSIVE enum_indexes(n) AS (
          SELECT 1
          UNION ALL
          SELECT n+1 FROM enum_indexes WHERE n < 6
        )
        SELECT enum_indexes.n,e.id,e.e_basic
        FROM enum_indexes
        LEFT JOIN split_type_enum_set.es_core AS e
          ON e.e_basic+0=enum_indexes.n
        ORDER BY enum_indexes.n,e.id;
SELECT id,e_basic,s_basic,
               ROW_NUMBER() OVER (
                 PARTITION BY e_basic ORDER BY s_basic+0,id
               ) AS rn,
               SUM(s_basic+0) OVER (
                 ORDER BY e_basic
                 ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
               ) AS running_mask
        FROM split_type_enum_set.es_core
        ORDER BY e_basic,rn;
SELECT LEAD(e_basic) OVER (ORDER BY e_basic)
        FROM split_type_enum_set.es_core
        GROUP BY e_basic WITH ROLLUP
        EXCEPT
        SELECT NULL;
