[dpc_range_routing_columns]
sql:
select id, note, sf_get_ep_seqno(rowid) as ep_seqno
from catalog1.app.dpc_range order by id
expect:
{
  "id" : [ "/catalog1/app/dpc_range/id/" ],
  "note" : [ "/catalog1/app/dpc_range/note/" ],
  "ep_seqno" : [ "/catalog1/app/dpc_range/rowid/" ]
}

{
  "ep_seqno" : [ "/catalog1/app/dpc_hash/rowid/" ],
