[ChParseColumnExpressionTest.test1]
sql:
select 1 + 1 as x
expect:
{
  "x" : []
}
----------
[ChParseColumnExpressionTest.test2]
sql:
select 1 + column1 as x from table1
expect:
{
  "x" : [ "/catalog1/schema1/table1/column1/" ]
}
----------
[ChParseColumnExpressionTest.test3]
sql:
select 1 + column1 + (select column1 from table2 limit 1) as x from table1
expect:
{
  "x" : [ "/catalog1/schema1/table1/column1/", "/catalog1/schema1/table2/column1/" ]
}
----------
[ChParseColumnExpressionTest.test4]
sql:
select 1 + column1 + (select column1 from table2 limit 1) + test(column2) as x from table1

{
  "x" : [ "/catalog1/schema1/table1/column1/", "/catalog1/schema1/table2/column1/", "/catalog1/schema1/table1/column2/" ]
}
----------
