[for_xml_path_columns]
sql:
select d.id,d.category from catalog1.app.xml_docs d order by id for xml path('row')
expect:
{
  "id": [
    "/catalog1/app/xml_docs/id/"
  ],
  "category": [
    "/catalog1/app/xml_docs/category/"
  ]
}
