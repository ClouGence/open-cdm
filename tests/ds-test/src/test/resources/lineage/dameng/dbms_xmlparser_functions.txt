declare
  p DBMS_XMLPARSER.PARSER;
  doc DBMS_XMLDOM.DOMDOCUMENT;
  doctype DBMS_XMLDOM.DOMDOCUMENTTYPE;
  c clob:='<?xml version="1.0"?><root><item>clob</item></root>';
  v varchar2(256);
  b boolean;
begin
  p:=DBMS_XMLPARSER.NEWPARSER(); DBMS_XMLPARSER.FREEPARSER(P=>p);
  v:=DBMS_XMLPARSER.GETBASEDIR(P=>p);
  doctype:=DBMS_XMLPARSER.GETDOCTYPE(P=>p);
  doc:=DBMS_XMLPARSER.GETDOCUMENTR(P=>p);
  doc:=DBMS_XMLPARSER.PARSE(URL=>'/tmp/a.xml');
  DBMS_XMLPARSER.PARSE(P=>p,URL=>'/tmp/a.xml');
  p:=DBMS_XMLPARSER.NEWPARSER(); DBMS_XMLPARSER.PARSEBUFFER(P=>p,DOC=>'<root/>'); DBMS_XMLPARSER.FREEPARSER(p);
  p:=DBMS_XMLPARSER.NEWPARSER(); DBMS_XMLPARSER.PARSECLOB(P=>p,DOC=>c); DBMS_XMLPARSER.FREEPARSER(p);
  DBMS_XMLPARSER.PARSEDTD(P=>p,URL=>'/tmp/a.dtd',ROOT=>'ROOT');
  DBMS_XMLPARSER.PARSEDTDBUFFER(P=>p,DTD=>'<!ELEMENT ROOT EMPTY>',ROOT=>'ROOT');
  c:='<!ELEMENT ROOT EMPTY>'; DBMS_XMLPARSER.PARSEDTDCLOB(P=>p,DTD=>c,ROOT=>'ROOT');
  v:=DBMS_XMLPARSER.RESETPARAMS();
  DBMS_XMLPARSER.SETBASEDIR(P=>p,DIR=>'/tmp');
  DBMS_XMLPARSER.SETDOCTYPE(P=>p,DTD=>doctype);
  DBMS_XMLPARSER.SETVALIDATIONMNODE(P=>p,YES=>TRUE);
  b:=DBMS_XMLPARSER.SETVALIDATIONMNODE(P=>p);
  p:=DBMS_XMLPARSER.NEWPARSER(); DBMS_XMLPARSER.FREEPARSER(p);
  p:=DBMS_XMLPARSER.NEWPARSER(); DBMS_XMLPARSER.PARSEBUFFER(p,'<root/>'); doc:=DBMS_XMLPARSER.GETDOCUMENT(P=>p); DBMS_XMLPARSER.FREEPARSER(p);
end;
