sql:
create schema app authorization SYSDBA
create table orders(id int primary key, amount decimal(12,2))
create view v_orders(order_id, order_amount) as select id, amount from orders

{
  "order_id" : [ "/catalog1/app/orders/id/" ],
  "order_amount" : [ "/catalog1/app/orders/amount/" ]
}
