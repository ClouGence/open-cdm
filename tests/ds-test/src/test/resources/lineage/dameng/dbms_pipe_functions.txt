[dbms_pipe_package_calls_no_static_lineage]
sql:
declare
  ret int;
  text_value varchar2(128);
  int_value int;
  bigint_value bigint;
  double_value double;
  number_value number;
  date_value date;
  rid rowid;
begin
  ret:=DBMS_PIPE.CREATE_PIPE(PIPENAME=>'CODEX_PIPE',MAXPIPESIZE=>8192,IS_PRIVATE=>TRUE);
  ret:=DBMS_PIPE.NEXT_ITEM_TYPE();
  DBMS_PIPE.PACK_MESSAGE(ITEM=>'TEXT');
  DBMS_PIPE.PACK_MESSAGE(ITEM=>CAST(1 AS INT));
  DBMS_PIPE.PACK_MESSAGE(ITEM=>CAST(2 AS BIGINT));
  DBMS_PIPE.PACK_MESSAGE(ITEM=>CAST(3.5 AS DOUBLE));
  DBMS_PIPE.PACK_MESSAGE(ITEM=>CAST(4.5 AS NUMBER));
  DBMS_PIPE.PACK_MESSAGE(ITEM=>CAST('2026-07-27' AS DATE));
  DBMS_PIPE.PACK_MESSAGE_ROWID(ITEM=>rid);
  DBMS_PIPE.PURGE(PIPENAME=>'CODEX_PIPE');
  ret:=DBMS_PIPE.RECEIVE_MESSAGE(PIPENAME=>'CODEX_PIPE',TIMEOUT=>0);
  ret:=DBMS_PIPE.REMOVE_PIPE(PIPENAME=>'CODEX_PIPE');
  DBMS_PIPE.RESET_BUFFER();
  ret:=DBMS_PIPE.SEND_MESSAGE(PIPENAME=>'CODEX_PIPE',TIMEOUT=>0,MAXPIPESIZE=>8192);
  text_value:=DBMS_PIPE.UNIQUE_SESSION_NAME();
  DBMS_PIPE.UNPACK_MESSAGE(ITEM=>text_value);
  DBMS_PIPE.UNPACK_MESSAGE(ITEM=>int_value);
  DBMS_PIPE.UNPACK_MESSAGE(ITEM=>bigint_value);
  DBMS_PIPE.UNPACK_MESSAGE(ITEM=>double_value);
  DBMS_PIPE.UNPACK_MESSAGE(ITEM=>number_value);
  DBMS_PIPE.UNPACK_MESSAGE(ITEM=>date_value);
  DBMS_PIPE.UNPACK_MESSAGE_ROWID(ITEM=>rid);
end;
expect:
{}
