[mpp_colocated_join_columns]
sql:
select a.id,a.note,b.note as b_note
from catalog1.app.mpp_hash_a a join catalog1.app.mpp_hash_b b on a.id=b.id
expect:
{
  "id" : [ "/catalog1/app/mpp_hash_a/id/" ],
  "note" : [ "/catalog1/app/mpp_hash_a/note/" ],
  "b_note" : [ "/catalog1/app/mpp_hash_b/note/" ]
}
