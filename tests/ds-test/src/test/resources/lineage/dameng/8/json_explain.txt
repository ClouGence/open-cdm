[explain_cte_columns]

explain as CTE_PLAN for
with q as(
  select t.id,t.v from catalog1.app.ex_target t
)
select q.id,q.v from q

{
  "id": ["/catalog1/app/ex_target/id/"],
  "v": ["/catalog1/app/ex_target/v/"]
}
