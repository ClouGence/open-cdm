select
  numtodsinterval(t.duration_value,'DAY') as ds_value,
  numtoyminterval(t.period_value,'MONTH') as ym_value,
  to_dsinterval(t.day_interval_text) as ds_text,
  to_yminterval(t.year_interval_text) as ym_text,
  overlaps(t.created_at,t.updated_at,t.active_at,t.archived_at) as overlap_value
from catalog1.app.events t

{

  "ym_text": ["/catalog1/app/events/year_interval_text/"],
