-- 包含子查询的跨Schema三表连接[1,7](@ref)
SELECT s1t1.column1 AS main_id,
       (SELECT MAX(s2t2.column3) 
        FROM schema2.table2 s2t2 
        WHERE s2t2.column4 = s1t1.column2) AS max_val,
       RANK() OVER (ORDER BY s1t2.column5 DESC) AS rank_level
FROM schema1.table1 s1t1
JOIN schema1.table2 s1t2 
  ON s1t1.column3 = s1t2.column1
LEFT JOIN schema2.table1 s2t1 
  ON s1t1.column4 = s2t1.column5
WHERE s2t1.column6 IN (
    SELECT DISTINCT column2 
    FROM schema2.table2 
    WHERE column4 > 100
);

  "rank_level" : [ "(6:29~6:41) /catalog1/schema1/table2/column5/" ]
}

SELECT s1t1.column2 AS category,
       SUM(s2t1.column4) AS total_sum,
       AVG(s1t2.column5) OVER (PARTITION BY s1t1.column2) AS avg_value,
       (SELECT COUNT(*) 
        FROM schema2.table2 
        WHERE column3 = s1t1.column3) AS related_count
FROM schema1.table1 s1t1
LEFT JOIN schema1.table2 s1t2 
  ON s1t1.column1 = s1t2.column6
RIGHT JOIN schema2.table1 s2t1 
  ON s1t1.column4 = s2t1.column3
GROUP BY s1t1.column2
HAVING total_sum > (
    SELECT AVG(column4)*1.5 
    FROM schema2.table1
);

[MyParseColumnComplexTest.test3]
sql:
(
    SELECT 
        s1t1.*,
        s2t2.column1 AS s2_col1,
        ROW_NUMBER() OVER (ORDER BY s1t2.column5) AS conflict_alias
    FROM schema1.table1 s1t1
    LEFT JOIN schema2.table2 s2t2 
      ON s1t1.column6 = s2t2.column3
    LEFT JOIN schema1.table2 s1t2 
      ON s1t1.column2 = s1t2.column4
) 
UNION
(
    SELECT 
        s2t1.*,
        s1t2.column2 AS s1_col2,
        RANK() OVER (PARTITION BY s2t1.column3 ORDER BY s2t1.column4) AS conflict_alias 
    FROM schema2.table1 s2t1
    RIGHT JOIN schema1.table2 s1t2 
      ON s2t1.column5 = s1t2.column1
    WHERE s2t1.column6 IN (
        SELECT MAX(column6) 
        FROM schema2.table2 
        GROUP BY column2
    )
)
ORDER BY 
    CASE 
        WHEN conflict_alias > 100 THEN column1 
        ELSE column2 
    END DESC;
