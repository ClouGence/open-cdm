          SELECT id,
                 JSON_VALUE(
                   doc,'$.kind' RETURNING CHAR(32)
                 ) AS kind,
                 JSON_VALUE(
                   doc,'$.score' RETURNING UNSIGNED
                   DEFAULT 0 ON EMPTY
