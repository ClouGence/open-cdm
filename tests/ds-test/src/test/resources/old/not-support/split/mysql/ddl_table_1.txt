CREATE TABLE `feedback` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `user_id` bigint NOT NULL DEFAULT '0' COMMENT '用户Id',
  `user_name` varchar(255) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '用户名称',
  `feedback_type` tinyint NOT NULL DEFAULT '1' COMMENT '反馈类型: 1 功能建议 2 闪退卡顿 3 其他',
  `feedback_detail` varchar(600) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '反馈详情',
  `pic_urls` varchar(512) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '相关图片地址',
  `phone` varchar(20) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '联系手机号',
  `feedback_source` tinyint NOT NULL DEFAULT '1' COMMENT '来源 0:患者APP,1:医生APP,2:患者微信',
  `feedback_logs` varchar(255) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '机型等日志',
  `created_at` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `created_by` varchar(16) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '创建人',
  `changed_at` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '修改时间',
  `changed_by` varchar(16) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '修改人',
  `is_deleted` tinyint(1) NOT NULL DEFAULT '0' COMMENT '逻辑删除',
  `version` int NOT NULL DEFAULT '0' COMMENT '乐观锁',
  `encrypt_mobile` varchar(64) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '手机密文',
  `feedback_sub_source` tinyint NOT NULL DEFAULT '0' COMMENT '子来源 0意见来源 1项目意见反馈',
  `project_code` varchar(255) COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '项目编码',
  PRIMARY KEY (`id`) USING BTREE,
  KEY ```user_id``` (`user_id`) USING BTREE,
  KEY `feedback_type` (`feedback_type`) USING BTREE,
  KEY `phone` (`phone`) USING BTREE,
  KEY `idx_encrypt_mobile` (`encrypt_mobile`)
) ENGINE=InnoDB AUTO_INCREMENT=2890 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci COMMENT='用户使用反馈表';
CREATE TABLE `users` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_password` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL,
  `email` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL,
  `mobile` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL,
  `nickname` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL,
  `gender` tinyint DEFAULT NULL COMMENT '0:男 1:女 2:保密',
  `realname` varchar(200) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '真实姓名',
  `birthday` datetime DEFAULT NULL,
  `user_role` int NOT NULL COMMENT '0：普通用户 1：Sales  2：IB',
  `nation` int NOT NULL COMMENT '手机区号',
  `introduction` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL,
  `avatar_url` mediumtext CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci,
  `cover_url` mediumtext CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci,
  `is_random_nick_name` bit(1) NOT NULL DEFAULT b'1',
  `invite_code` int NOT NULL,
  `parent_invite_code` int NOT NULL,
  `country_code` varchar(2) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '2位国家代码',
  `account_status` tinyint NOT NULL COMMENT '0:正常 2:禁用 3:注销',
  `last_login_date` bigint NOT NULL COMMENT '上次登录时间',
  `last_activity_date` bigint NOT NULL COMMENT '上次活跃时间',
  `last_login_ip` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '上次登录ip',
  `last_login_region` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '上次登录地区country-province-city',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `backup_contact` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT '' COMMENT '备用联系方式(email,mobile)',
  `national_code` varchar(2) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '国藉(2位国家代码)',
  `login_name` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '登入名',
  `mobile_verified` bit(1) NOT NULL DEFAULT b'1' COMMENT 'Mobile是否已经OTP验证',
  `email_verified` bit(1) NOT NULL DEFAULT b'1' COMMENT 'Email是否已经已经OTP验证',
  `user_alias` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT '' COMMENT '别名',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `ix_email` (`email`) USING BTREE,
  KEY `ix_mobile` (`mobile`) USING BTREE,
  KEY `ix_nickname` (`nickname`) USING BTREE,
  KEY `ix_users_logindate` (`last_login_date`) USING BTREE,
  KEY `ix_users_lastactivitydate` (`last_activity_date`) USING BTREE,
  KEY `users_ix_login_name` (`login_name`) USING BTREE
) ENGINE=InnoDB AUTO_INCREMENT=308984 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci ROW_FORMAT=DYNAMIC;
