[split_block_dameng_trigger_table_view_0_015_behavior_review]
sql:
create or replace trigger trg_all_dml
after insert or delete or update on trg_target
for each row
declare
  v_operation varchar(10);
  v_old_id int;
  v_new_id int;
begin
  if inserting then
    v_operation = 'INSERT';
    v_new_id = :new.id;
  elseif updating then
    v_operation = 'UPDATE';
    v_old_id = :old.id;
    v_new_id = :new.id;
  else
    v_operation = 'DELETE';
    v_old_id = :old.id;
  end if;
  insert into trg_log(trigger_name, operation, old_id, new_id)
  values('TRG_ALL_DML', v_operation, v_old_id, v_new_id);
end;
